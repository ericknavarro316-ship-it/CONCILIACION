Option Explicit

Public Sub I01_CONCILIAR_VENTAS_BBVA_VS_BANCOS()

    Const SH_VENTAS As String = "VENTAS_BBVA"

    Dim wsVentas As Worksheet
    Dim hdrVentas As Long, lastVentas As Long
    Dim colVId As Long, colVUUID As Long, colVCuenta As Long
    Dim colVTotal As Long, colVImporte As Long, colVMonto As Long
    Dim colVPrecio As Long, colVCantidad As Long

    Dim wsCFDIPUE As Worksheet, wsCFDIPPD As Worksheet
    Dim hdrCFDIPUE As Long, hdrCFDIPPD As Long
    Dim colPUEUUID As Long, colPUECuenta As Long, colPUEId As Long
    Dim colPPDUUID As Long, colPPDCuenta As Long, colPPDId As Long

    Dim dictSalesByAmount As Object
    Dim dictGroupRows As Object
    Dim dictGroupUUID As Object
    Dim dictGroupAmount As Object
    Dim dictGroupCuenta As Object

    Dim dictPUE As Object, dictPPD As Object

    Dim ws As Worksheet
    Dim matchedCount As Long

    Dim prevCalc As XlCalculation
    Dim prevScreen As Boolean
    Dim prevEvents As Boolean

    On Error GoTo ErrHandler

    Set wsVentas = I01_GetSheet(SH_VENTAS)
    If wsVentas Is Nothing Then
        MsgBox "No existe la hoja '" & SH_VENTAS & "'.", vbExclamation
        Exit Sub
    End If

    hdrVentas = I01_FindHeaderRowVentas(wsVentas)
    If hdrVentas = 0 Then
        MsgBox "No pude identificar la fila de encabezados en VENTAS_BBVA.", vbExclamation
        Exit Sub
    End If

    colVId = I01_FindHeaderCol(wsVentas, hdrVentas, Array("ID_venta", "id_venta"))
    colVUUID = I01_FindHeaderCol(wsVentas, hdrVentas, Array("UUID", "uuid"))
    colVCuenta = I01_EnsureHeaderCol(wsVentas, hdrVentas, "Cuenta bancaria")

    colVTotal = I01_FindHeaderCol(wsVentas, hdrVentas, Array("Total", "total"))
    colVImporte = I01_FindHeaderCol(wsVentas, hdrVentas, Array("Importe", "importe"))
    colVMonto = I01_FindHeaderCol(wsVentas, hdrVentas, Array("Monto", "monto", "Monto total", "monto total", "Total venta", "total venta", "Valor de la operación", "Valor de la operacion"))
    colVPrecio = I01_FindHeaderCol(wsVentas, hdrVentas, Array("precio_unitario", "precio unitario", "Precio_Unitario", "Precio Unitario"))
    colVCantidad = I01_FindHeaderCol(wsVentas, hdrVentas, Array("Cantidad", "cantidad", "Qty", "qty", "Cant", "cant"))

    If colVId = 0 Or colVUUID = 0 Then
        MsgBox "Faltan columnas requeridas en VENTAS_BBVA." & vbCrLf & _
               "Se requiere: ID_venta y UUID.", vbExclamation
        Exit Sub
    End If

    If colVTotal = 0 And colVImporte = 0 And colVMonto = 0 And colVPrecio = 0 Then
        MsgBox "No encontré una columna de importe en VENTAS_BBVA." & vbCrLf & _
               "Busqué: Total, Importe, Monto y Precio_Unitario.", vbExclamation
        Exit Sub
    End If

    lastVentas = I01_LastUsedRow(wsVentas)
    If lastVentas <= hdrVentas Then
        MsgBox "VENTAS_BBVA no tiene datos.", vbExclamation
        Exit Sub
    End If

    prevCalc = Application.Calculation
    prevScreen = Application.ScreenUpdating
    prevEvents = Application.EnableEvents

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    ' ==========================================
    ' Preparar índices CFDI I PUE / PPD por UUID
    ' ==========================================
    Set wsCFDIPUE = I01_GetSheet("CFDI_I_PUE")
    If Not wsCFDIPUE Is Nothing Then
        hdrCFDIPUE = I01_FindHeaderRowUUID(wsCFDIPUE)
        If hdrCFDIPUE > 0 Then
            colPUEUUID = I01_FindHeaderCol(wsCFDIPUE, hdrCFDIPUE, Array("UUID", "uuid"))
            colPUECuenta = I01_EnsureHeaderCol(wsCFDIPUE, hdrCFDIPUE, "Cuenta bancaria")
            colPUEId = I01_EnsureHeaderColMulti(wsCFDIPUE, hdrCFDIPUE, Array("ID_venta", "id_venta"), "ID_venta")

            Set dictPUE = CreateObject("Scripting.Dictionary")
            I01_BuildUUIDIndex wsCFDIPUE, hdrCFDIPUE, colPUEUUID, dictPUE
        End If
    End If

    Set wsCFDIPPD = I01_GetSheet("CFDI_I_PPD")
    If Not wsCFDIPPD Is Nothing Then
        hdrCFDIPPD = I01_FindHeaderRowUUID(wsCFDIPPD)
        If hdrCFDIPPD > 0 Then
            colPPDUUID = I01_FindHeaderCol(wsCFDIPPD, hdrCFDIPPD, Array("UUID", "uuid"))
            colPPDCuenta = I01_EnsureHeaderCol(wsCFDIPPD, hdrCFDIPPD, "Cuenta bancaria")
            colPPDId = I01_EnsureHeaderColMulti(wsCFDIPPD, hdrCFDIPPD, Array("ID_venta", "id_venta"), "ID_venta")

            Set dictPPD = CreateObject("Scripting.Dictionary")
            I01_BuildUUIDIndex wsCFDIPPD, hdrCFDIPPD, colPPDUUID, dictPPD
        End If
    End If

    ' ==========================================
    ' Agrupar VENTAS_BBVA por ID_venta
    ' ==========================================
    Set dictSalesByAmount = CreateObject("Scripting.Dictionary")
    Set dictGroupRows = CreateObject("Scripting.Dictionary")
    Set dictGroupUUID = CreateObject("Scripting.Dictionary")
    Set dictGroupAmount = CreateObject("Scripting.Dictionary")
    Set dictGroupCuenta = CreateObject("Scripting.Dictionary")

    I01_BuildVentasGroups wsVentas, hdrVentas, lastVentas, _
                          colVId, colVUUID, colVCuenta, _
                          colVTotal, colVImporte, colVMonto, colVPrecio, colVCantidad, _
                          dictSalesByAmount, dictGroupRows, dictGroupUUID, dictGroupAmount, dictGroupCuenta

    ' ==========================================
    ' Recorrer hojas bancarias numéricas
    ' ==========================================
    For Each ws In ThisWorkbook.Worksheets
        If I01_EsHojaNumerica(ws.Name) Then
            If Not I01_SheetHasText(ws, "SIN MOVIMIENTOS") Then
                matchedCount = matchedCount + I01_ProcesarHojaBanco( _
                    ws, wsVentas, _
                    dictSalesByAmount, dictGroupRows, dictGroupUUID, dictGroupCuenta, _
                    colVCuenta, _
                    wsCFDIPUE, dictPUE, colPUECuenta, colPUEId, _
                    wsCFDIPPD, dictPPD, colPPDCuenta, colPPDId)
            End If
        End If
    Next ws

    Application.Calculation = prevCalc
    Application.EnableEvents = prevEvents
    Application.ScreenUpdating = prevScreen

    MsgBox "Conciliación VENTAS_BBVA vs bancos terminada." & vbCrLf & _
           "Ventas conciliadas: " & matchedCount, vbInformation
    Exit Sub

ErrHandler:
    Application.Calculation = prevCalc
    Application.EnableEvents = prevEvents
    Application.ScreenUpdating = prevScreen

    MsgBox "Error en I01_CONCILIAR_VENTAS_BBVA_VS_BANCOS: " & Err.Description, vbExclamation
End Sub


Private Sub I01_BuildVentasGroups( _
    ByVal wsVentas As Worksheet, _
    ByVal hdrRow As Long, _
    ByVal lastRow As Long, _
    ByVal colVId As Long, _
    ByVal colVUUID As Long, _
    ByVal colVCuenta As Long, _
    ByVal colVTotal As Long, _
    ByVal colVImporte As Long, _
    ByVal colVMonto As Long, _
    ByVal colVPrecio As Long, _
    ByVal colVCantidad As Long, _
    ByVal dictSalesByAmount As Object, _
    ByVal dictGroupRows As Object, _
    ByVal dictGroupUUID As Object, _
    ByVal dictGroupAmount As Object, _
    ByVal dictGroupCuenta As Object)

    Dim r As Long
    Dim idVenta As String, uuidTxt As String, cuentaTxt As String
    Dim amountLine As Double
    Dim cents As Long
    Dim coll As Collection
    Dim keyAmount As String

    For r = hdrRow + 1 To lastRow

        idVenta = Trim$(CStr(wsVentas.Cells(r, colVId).Value))
        If idVenta = "" Then GoTo SiguienteR

        uuidTxt = Trim$(CStr(wsVentas.Cells(r, colVUUID).Value))
        cuentaTxt = Trim$(CStr(wsVentas.Cells(r, colVCuenta).Value))
        amountLine = I01_GetVentaLineAmount(wsVentas, r, colVTotal, colVImporte, colVMonto, colVPrecio, colVCantidad)

        If amountLine = 0 Then GoTo SiguienteR

        If Not dictGroupRows.Exists(idVenta) Then
            Set coll = New Collection
            dictGroupRows.Add idVenta, coll
            dictGroupAmount.Add idVenta, CLng(0)
            dictGroupUUID.Add idVenta, uuidTxt
            dictGroupCuenta.Add idVenta, cuentaTxt
        End If

        Set coll = dictGroupRows(idVenta)
        coll.Add r

        dictGroupAmount(idVenta) = CLng(dictGroupAmount(idVenta)) + I01_ToCents(amountLine)

        If Trim$(CStr(dictGroupUUID(idVenta))) = "" And uuidTxt <> "" Then
            dictGroupUUID(idVenta) = uuidTxt
        End If

        If Trim$(CStr(dictGroupCuenta(idVenta))) = "" And cuentaTxt <> "" Then
            dictGroupCuenta(idVenta) = cuentaTxt
        End If

SiguienteR:
    Next r

    ' Solo grupos sin cuenta bancaria previa
    Dim idKey As Variant
    For Each idKey In dictGroupAmount.Keys
        If Trim$(CStr(dictGroupCuenta(idKey))) = "" Then
            cents = CLng(dictGroupAmount(idKey))
            If cents <> 0 Then
                keyAmount = CStr(cents)
                If Not dictSalesByAmount.Exists(keyAmount) Then
                    Set coll = New Collection
                    dictSalesByAmount.Add keyAmount, coll
                End If
                Set coll = dictSalesByAmount(keyAmount)
                coll.Add CStr(idKey)
            End If
        End If
    Next idKey
End Sub


Private Function I01_GetVentaLineAmount( _
    ByVal ws As Worksheet, _
    ByVal rowNum As Long, _
    ByVal colVTotal As Long, _
    ByVal colVImporte As Long, _
    ByVal colVMonto As Long, _
    ByVal colVPrecio As Long, _
    ByVal colVCantidad As Long) As Double

    Dim totalVal As Double
    Dim importeVal As Double
    Dim montoVal As Double
    Dim precioVal As Double
    Dim cantidadVal As Double

    If colVTotal > 0 Then
        totalVal = I01_ParseNumber(ws.Cells(rowNum, colVTotal).Value)
        If totalVal <> 0 Then
            I01_GetVentaLineAmount = totalVal
            Exit Function
        End If
    End If

    If colVImporte > 0 Then
        importeVal = I01_ParseNumber(ws.Cells(rowNum, colVImporte).Value)
        If importeVal <> 0 Then
            I01_GetVentaLineAmount = importeVal
            Exit Function
        End If
    End If

    If colVMonto > 0 Then
        montoVal = I01_ParseNumber(ws.Cells(rowNum, colVMonto).Value)
        If montoVal <> 0 Then
            I01_GetVentaLineAmount = montoVal
            Exit Function
        End If
    End If

    If colVPrecio > 0 Then
        precioVal = I01_ParseNumber(ws.Cells(rowNum, colVPrecio).Value)

        If colVCantidad > 0 Then
            cantidadVal = I01_ParseNumber(ws.Cells(rowNum, colVCantidad).Value)
            If cantidadVal = 0 Then cantidadVal = 1
            I01_GetVentaLineAmount = precioVal * cantidadVal
        Else
            I01_GetVentaLineAmount = precioVal
        End If
    End If
End Function


Private Function I01_ProcesarHojaBanco( _
    ByVal ws As Worksheet, _
    ByVal wsVentas As Worksheet, _
    ByVal dictSalesByAmount As Object, _
    ByVal dictGroupRows As Object, _
    ByVal dictGroupUUID As Object, _
    ByVal dictGroupCuenta As Object, _
    ByVal colVCuenta As Long, _
    ByVal wsCFDIPUE As Worksheet, _
    ByVal dictPUE As Object, _
    ByVal colPUECuenta As Long, _
    ByVal colPUEId As Long, _
    ByVal wsCFDIPPD As Worksheet, _
    ByVal dictPPD As Object, _
    ByVal colPPDCuenta As Long, _
    ByVal colPPDId As Long) As Long

    Dim hdrDia As Range, hdrCargo As Range, hdrAbono As Range
    Dim hdrRow As Long
    Dim colAbono As Long, colCargo As Long, colObs As Long, colIdVenta As Long
    Dim lastRow As Long, r As Long

    Dim abonoVal As Double, cargoVal As Double
    Dim cents As Long, tol As Long
    Dim idVenta As String, uuidTxt As String
    Dim obsTxt As String, idTxt As String

    Set hdrDia = ws.Cells.Find(What:="Día", LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    If hdrDia Is Nothing Then
        Set hdrDia = ws.Cells.Find(What:="Dia", LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    End If
    If hdrDia Is Nothing Then Exit Function

    hdrRow = hdrDia.Row

    Set hdrCargo = I01_FindHeader(ws, hdrRow, Array("Cargo"))
    Set hdrAbono = I01_FindHeader(ws, hdrRow, Array("Abono"))

    If hdrCargo Is Nothing Or hdrAbono Is Nothing Then Exit Function

    colCargo = hdrCargo.Column
    colAbono = hdrAbono.Column
    colObs = I01_EnsureHeaderCol(ws, hdrRow, "OBS")
    colIdVenta = I01_EnsureHeaderColMulti(ws, hdrRow, Array("ID_venta", "id_venta"), "ID_venta")

    lastRow = I01_LastUsedRow(ws)
    If lastRow <= hdrRow Then Exit Function

    For r = hdrRow + 1 To lastRow

        abonoVal = I01_ParseNumber(ws.Cells(r, colAbono).Value)
        cargoVal = I01_ParseNumber(ws.Cells(r, colCargo).Value)

        If abonoVal <= 0 Then GoTo SiguienteR
        If cargoVal <> 0 Then GoTo SiguienteR

        obsTxt = Trim$(CStr(ws.Cells(r, colObs).Value))
        idTxt = Trim$(CStr(ws.Cells(r, colIdVenta).Value))

        If obsTxt <> "" Then GoTo SiguienteR
        If idTxt <> "" Then GoTo SiguienteR

        cents = I01_ToCents(abonoVal)
        tol = I01_ToleranceCents(cents)

        If I01_TryPopVentaGroupMatch(dictSalesByAmount, cents, tol, idVenta) Then

            uuidTxt = Trim$(CStr(dictGroupUUID(idVenta)))

            ws.Cells(r, colIdVenta).Value = idVenta
            ws.Cells(r, colObs).Value = uuidTxt

            I01_UpdateVentasGroupCuenta wsVentas, dictGroupRows(idVenta), colVCuenta, ws.Name

            If Not wsCFDIPUE Is Nothing Then
                I01_UpdateCFDIByUUID wsCFDIPUE, dictPUE, uuidTxt, colPUECuenta, colPUEId, ws.Name, idVenta
            End If

            If Not wsCFDIPPD Is Nothing Then
                I01_UpdateCFDIByUUID wsCFDIPPD, dictPPD, uuidTxt, colPPDCuenta, colPPDId, ws.Name, idVenta
            End If

            I01_ProcesarHojaBanco = I01_ProcesarHojaBanco + 1
        End If

SiguienteR:
    Next r

    ws.Columns(colObs).NumberFormat = "@"
    ws.Columns(colIdVenta).NumberFormat = "@"
End Function


Private Sub I01_UpdateVentasGroupCuenta( _
    ByVal wsVentas As Worksheet, _
    ByVal rowsColl As Collection, _
    ByVal colVCuenta As Long, _
    ByVal cuentaBancaria As String)

    Dim i As Long
    Dim rr As Long

    For i = 1 To rowsColl.Count
        rr = CLng(rowsColl(i))
        wsVentas.Cells(rr, colVCuenta).Value = cuentaBancaria
    Next i
End Sub


Private Sub I01_UpdateCFDIByUUID( _
    ByVal ws As Worksheet, _
    ByVal dictUUIDRows As Object, _
    ByVal uuidValue As String, _
    ByVal colCuenta As Long, _
    ByVal colIdVenta As Long, _
    ByVal cuentaBancaria As String, _
    ByVal idVenta As String)

    Dim k As String
    Dim coll As Collection
    Dim i As Long, rr As Long

    If dictUUIDRows Is Nothing Then Exit Sub

    k = I01_NormKey(uuidValue)
    If k = "" Then Exit Sub
    If Not dictUUIDRows.Exists(k) Then Exit Sub

    Set coll = dictUUIDRows(k)

    For i = 1 To coll.Count
        rr = CLng(coll(i))
        ws.Cells(rr, colCuenta).Value = cuentaBancaria
        ws.Cells(rr, colIdVenta).Value = idVenta
    Next i
End Sub


Private Sub I01_BuildUUIDIndex( _
    ByVal ws As Worksheet, _
    ByVal hdrRow As Long, _
    ByVal colUUID As Long, _
    ByVal dictUUIDRows As Object)

    Dim lastRow As Long, r As Long
    Dim k As String
    Dim coll As Collection

    lastRow = I01_LastUsedRow(ws)
    If lastRow <= hdrRow Then Exit Sub

    For r = hdrRow + 1 To lastRow
        k = I01_NormKey(ws.Cells(r, colUUID).Value)
        If k <> "" Then
            If Not dictUUIDRows.Exists(k) Then
                Set coll = New Collection
                dictUUIDRows.Add k, coll
            End If
            Set coll = dictUUIDRows(k)
            coll.Add r
        End If
    Next r
End Sub


Private Function I01_TryPopVentaGroupMatch( _
    ByVal dictSalesByAmount As Object, _
    ByVal centsBase As Long, _
    ByVal tolCents As Long, _
    ByRef idVenta As String) As Boolean

    Dim d As Long
    Dim k As String
    Dim coll As Collection

    For d = 0 To tolCents

        k = CStr(centsBase + d)
        If dictSalesByAmount.Exists(k) Then
            Set coll = dictSalesByAmount(k)
            If coll.Count > 0 Then
                idVenta = CStr(coll(1))
                coll.Remove 1
                If coll.Count = 0 Then dictSalesByAmount.Remove k
                I01_TryPopVentaGroupMatch = True
                Exit Function
            End If
        End If

        If d > 0 Then
            k = CStr(centsBase - d)
            If dictSalesByAmount.Exists(k) Then
                Set coll = dictSalesByAmount(k)
                If coll.Count > 0 Then
                    idVenta = CStr(coll(1))
                    coll.Remove 1
                    If coll.Count = 0 Then dictSalesByAmount.Remove k
                    I01_TryPopVentaGroupMatch = True
                    Exit Function
                End If
            End If
        End If
    Next d
End Function


Private Function I01_FindHeaderRowVentas(ByVal ws As Worksheet) As Long
    Dim r As Long

    For r = 1 To 10
        If I01_FindHeaderCol(ws, r, Array("ID_venta", "id_venta")) > 0 And _
           I01_FindHeaderCol(ws, r, Array("UUID", "uuid")) > 0 Then
            I01_FindHeaderRowVentas = r
            Exit Function
        End If
    Next r
End Function


Private Function I01_FindHeaderRowUUID(ByVal ws As Worksheet) As Long
    Dim r As Long

    For r = 1 To 10
        If I01_FindHeaderCol(ws, r, Array("UUID", "uuid")) > 0 Then
            I01_FindHeaderRowUUID = r
            Exit Function
        End If
    Next r
End Function


Private Function I01_GetSheet(ByVal sheetName As String) As Worksheet
    On Error Resume Next
    Set I01_GetSheet = ThisWorkbook.Worksheets(sheetName)
    On Error GoTo 0
End Function


Private Function I01_FindHeader(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headers As Variant) As Range
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim txt As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        txt = I01_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headers) To UBound(headers)
            If txt = I01_NormHeader(CStr(headers(i))) Then
                Set I01_FindHeader = ws.Cells(hdrRow, c)
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function I01_FindHeaderCol(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headerNames As Variant) As Long
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim cellText As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        cellText = I01_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headerNames) To UBound(headerNames)
            If cellText = I01_NormHeader(CStr(headerNames(i))) Then
                I01_FindHeaderCol = c
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function I01_EnsureHeaderCol(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headerText As String) As Long
    Dim col As Long

    col = I01_FindHeaderCol(ws, hdrRow, Array(headerText))

    If col > 0 Then
        I01_EnsureHeaderCol = col
    Else
        I01_EnsureHeaderCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column + 1
        ws.Cells(hdrRow, I01_EnsureHeaderCol).Value = headerText
        ws.Cells(hdrRow, I01_EnsureHeaderCol).Font.Bold = True
    End If
End Function


Private Function I01_EnsureHeaderColMulti(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headers As Variant, ByVal newHeader As String) As Long
    Dim col As Long

    col = I01_FindHeaderCol(ws, hdrRow, headers)

    If col > 0 Then
        I01_EnsureHeaderColMulti = col
    Else
        I01_EnsureHeaderColMulti = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column + 1
        ws.Cells(hdrRow, I01_EnsureHeaderColMulti).Value = newHeader
        ws.Cells(hdrRow, I01_EnsureHeaderColMulti).Font.Bold = True
    End If
End Function


Private Function I01_NormHeader(ByVal s As String) As String
    s = UCase$(Trim$(s))
    s = Replace(s, "Á", "A")
    s = Replace(s, "É", "E")
    s = Replace(s, "Í", "I")
    s = Replace(s, "Ó", "O")
    s = Replace(s, "Ú", "U")
    s = Replace(s, "Ü", "U")
    s = Replace(s, ".", "")
    s = Replace(s, ",", "")
    s = Replace(s, "/", "")
    Do While InStr(s, "  ") > 0
        s = Replace(s, "  ", " ")
    Loop
    I01_NormHeader = s
End Function


Private Function I01_LastUsedRow(ByVal ws As Worksheet) As Long
    Dim c As Range

    Set c = ws.Cells.Find(What:="*", _
                          After:=ws.Range("A1"), _
                          LookIn:=xlFormulas, _
                          LookAt:=xlPart, _
                          SearchOrder:=xlByRows, _
                          SearchDirection:=xlPrevious, _
                          MatchCase:=False)

    If c Is Nothing Then
        I01_LastUsedRow = 0
    Else
        I01_LastUsedRow = c.Row
    End If
End Function


Private Function I01_ParseNumber(ByVal v As Variant) As Double
    Dim s As String
    Dim neg As Boolean

    On Error GoTo ErrHandler

    If IsNumeric(v) Then
        I01_ParseNumber = CDbl(v)
        Exit Function
    End If

    s = Trim$(CStr(v))
    If s = "" Then Exit Function

    s = Replace(s, Chr(160), "")
    s = Replace(s, "$", "")
    s = Replace(s, ",", "")
    s = Replace(s, " ", "")

    If Left$(s, 1) = "(" And Right$(s, 1) = ")" Then
        neg = True
        s = Replace(s, "(", "")
        s = Replace(s, ")", "")
    End If

    If IsNumeric(s) Then
        I01_ParseNumber = CDbl(s)
        If neg Then I01_ParseNumber = -I01_ParseNumber
    End If
    Exit Function

ErrHandler:
    I01_ParseNumber = 0
End Function


Private Function I01_ToCents(ByVal v As Variant) As Long
    I01_ToCents = CLng(Round(I01_ParseNumber(v) * 100, 0))
End Function


Private Function I01_ToleranceCents(ByVal centsAmount As Long) As Long
    If Abs(centsAmount) < 5000000 Then
        I01_ToleranceCents = 10
    Else
        I01_ToleranceCents = 50
    End If
End Function


Private Function I01_SheetHasText(ByVal ws As Worksheet, ByVal pattern As String) As Boolean
    Dim f As Range
    Set f = ws.Cells.Find(What:=pattern, LookIn:=xlValues, LookAt:=xlPart, MatchCase:=False)
    I01_SheetHasText = Not (f Is Nothing)
End Function


Private Function I01_EsHojaNumerica(ByVal nombreHoja As String) As Boolean
    I01_EsHojaNumerica = IsNumeric(nombreHoja)
End Function


Private Function I01_NormKey(ByVal v As Variant) As String
    Dim s As String

    s = Trim$(CStr(v))
    If s = "" Then Exit Function

    s = UCase$(s)
    s = Replace(s, Chr(160), "")
    s = Replace(s, " ", "")
    s = Replace(s, "-", "")
    s = Replace(s, "_", "")
    s = Replace(s, ".", "")

    I01_NormKey = s
End Function
