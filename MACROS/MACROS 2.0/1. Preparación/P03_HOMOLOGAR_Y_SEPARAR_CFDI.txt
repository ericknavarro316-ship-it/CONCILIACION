Option Explicit

Public Sub P03_HOMOLOGAR_Y_SEPARAR_CFDI

    Dim prevCalc As XlCalculation
    Dim prevScreen As Boolean
    Dim prevEvents As Boolean

    On Error GoTo ErrHandler

    prevCalc = Application.Calculation
    prevScreen = Application.ScreenUpdating
    prevEvents = Application.EnableEvents

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    ' CFDI E: usa nombre del EMISOR como contraparte
    C01X_ProcesarCFDI _
        sourceSheetName:="CFDI E", _
        prefixName:="CFDI_E", _
        nombreHeaders:=Array("Nombre, denominación o razón social del Emisor", "Nombre, denominacion o razon social del Emisor"), _
        usoHeaders:=Array("Uso CFDI Receptor"), _
        tipoHeaders:=Array("Tipo Comp.", "Tipo.Com"), _
        formaHeaders:=Array("Forma de Pago", "Forma Pago"), _
        metodoHeaders:=Array("Método de Pago", "Metodo de Pago", "METODO DE PAGO")

    ' CFDI I: usa nombre del RECEPTOR
    C01X_ProcesarCFDI _
        sourceSheetName:="CFDI I", _
        prefixName:="CFDI_I", _
        nombreHeaders:=Array("Nombre, denominación o razón social del Receptor", "Nombre, denominacion o razon social del Receptor"), _
        usoHeaders:=Array("Uso CFDI Receptor"), _
        tipoHeaders:=Array("Tipo.Com", "Tipo Comp."), _
        formaHeaders:=Array("Forma de Pago", "Forma Pago"), _
        metodoHeaders:=Array("Método de Pago", "Metodo de Pago", "METODO DE PAGO")

    Application.Calculation = prevCalc
    Application.EnableEvents = prevEvents
    Application.ScreenUpdating = prevScreen

    MsgBox "CFDI E / CFDI I preparados con el nuevo formato.", vbInformation
    Exit Sub

ErrHandler:
    Application.Calculation = prevCalc
    Application.EnableEvents = prevEvents
    Application.ScreenUpdating = prevScreen

    MsgBox "Error en C01_PREPARAR_CFDI_E_I_FORMATO2: " & Err.Description, vbExclamation
End Sub


Private Sub C01X_ProcesarCFDI( _
    ByVal sourceSheetName As String, _
    ByVal prefixName As String, _
    ByVal nombreHeaders As Variant, _
    ByVal usoHeaders As Variant, _
    ByVal tipoHeaders As Variant, _
    ByVal formaHeaders As Variant, _
    ByVal metodoHeaders As Variant)

    Dim wsSrc As Worksheet
    Dim wsL As Worksheet, wsPUE As Worksheet, wsPPD As Worksheet

    Dim hdrRow As Long, lastRow As Long, lastCol As Long
    Dim colUUID As Long, colFechaEmi As Long, colPDF As Long, colRFC As Long
    Dim colNombre As Long, colUso As Long, colTipo As Long
    Dim colForma As Long, colMetodo As Long, colSubTotal As Long
    Dim colISR As Long, colIVA As Long, colIVA16 As Long
    Dim colTotal As Long, colConceptos As Long, colEstatus As Long

    Dim mapL As Object, mapPUE As Object, mapPPD As Object
    Dim r As Long
    Dim outL As Long, outP As Long, outD As Long

    Dim uuid As String, rfc As String, nombre As String, usoCFDI As String
    Dim tipoComp As String, formaPago As String, metodoPago As String
    Dim pdfTxt As String, estatus As String, conceptos As String
    Dim subTotal As Double, isrVal As Double, ivaVal As Double, iva16Val As Double, totalVal As Double
    Dim dt As Date

    Set wsSrc = C01X_GetSheet(sourceSheetName)
    If wsSrc Is Nothing Then
        MsgBox "No existe la hoja '" & sourceSheetName & "'.", vbExclamation
        Exit Sub
    End If

    hdrRow = C01X_FindHeaderRowCFDI(wsSrc)
    If hdrRow = 0 Then
        MsgBox "No pude identificar la fila de encabezados en '" & sourceSheetName & "'.", vbExclamation
        Exit Sub
    End If

    colUUID = C01X_FindHeaderCol(wsSrc, hdrRow, Array("UUID"))
    colFechaEmi = C01X_FindHeaderCol(wsSrc, hdrRow, Array("Fecha Emisión", "Fecha Emision"))
    colPDF = C01X_FindHeaderCol(wsSrc, hdrRow, Array("Fecha Timbrado"))
    colRFC = C01X_FindHeaderCol(wsSrc, hdrRow, Array("RFC"))
    colNombre = C01X_FindHeaderCol(wsSrc, hdrRow, nombreHeaders)
    colUso = C01X_FindHeaderCol(wsSrc, hdrRow, usoHeaders)
    colTipo = C01X_FindHeaderCol(wsSrc, hdrRow, tipoHeaders)
    colForma = C01X_FindHeaderCol(wsSrc, hdrRow, formaHeaders)
    colMetodo = C01X_FindHeaderCol(wsSrc, hdrRow, metodoHeaders)
    colSubTotal = C01X_FindHeaderCol(wsSrc, hdrRow, Array("SubTotal XML", "Subtotal XML"))
    colISR = C01X_FindHeaderCol(wsSrc, hdrRow, Array("ISR", "ISR "))
    colIVA = C01X_FindHeaderCol(wsSrc, hdrRow, Array("IVA"))
    colIVA16 = C01X_FindHeaderCol(wsSrc, hdrRow, Array("IVA 16"))
    colTotal = C01X_FindHeaderCol(wsSrc, hdrRow, Array("Total"))
    colConceptos = C01X_FindHeaderCol(wsSrc, hdrRow, Array("Conceptos"))
    colEstatus = C01X_FindHeaderCol(wsSrc, hdrRow, Array("Estatus"))

    If colUUID = 0 Or colFechaEmi = 0 Or colRFC = 0 Or colNombre = 0 Or colMetodo = 0 Then
        MsgBox "Faltan encabezados requeridos en '" & sourceSheetName & "'." & vbCrLf & _
               "Se requiere al menos: UUID, Fecha Emisión, RFC, Nombre y Método de Pago.", vbExclamation
        Exit Sub
    End If

    lastRow = C01X_LastUsedRow(wsSrc)
    lastCol = C01X_LastUsedCol(wsSrc, hdrRow)

    If lastRow <= hdrRow Then Exit Sub

    ' Cargar cuenta bancaria previa
    Set mapL = CreateObject("Scripting.Dictionary")
    Set mapPUE = C01X_LoadCuentaMap(prefixName & "_PUE")
    Set mapPPD = C01X_LoadCuentaMap(prefixName & "_PPD")

    C01X_MergeMaps mapL, mapPUE
    C01X_MergeMaps mapL, mapPPD
    C01X_MergeMaps mapL, C01X_LoadCuentaMap(prefixName & "_LIMPIO")

    ' Recrear hojas
    C01X_DeleteSheetIfExists prefixName & "_LIMPIO"
    C01X_DeleteSheetIfExists prefixName & "_PUE"
    C01X_DeleteSheetIfExists prefixName & "_PPD"

    Set wsL = ThisWorkbook.Worksheets.Add(After:=wsSrc)
    wsL.Name = prefixName & "_LIMPIO"

    Set wsPUE = ThisWorkbook.Worksheets.Add(After:=wsL)
    wsPUE.Name = prefixName & "_PUE"

    Set wsPPD = ThisWorkbook.Worksheets.Add(After:=wsPUE)
    wsPPD.Name = prefixName & "_PPD"

    C01X_WriteHeaders wsL
    C01X_WriteHeaders wsPUE
    C01X_WriteHeaders wsPPD

    outL = 2
    outP = 2
    outD = 2

    For r = hdrRow + 1 To lastRow

        If C01X_RowIsEmpty(wsSrc, r, lastCol) Then GoTo NextR

        uuid = Trim$(CStr(wsSrc.Cells(r, colUUID).Text))
        If uuid = "" Then GoTo NextR

        If colRFC > 0 Then rfc = Trim$(CStr(wsSrc.Cells(r, colRFC).Text)) Else rfc = ""
        If colNombre > 0 Then nombre = Trim$(CStr(wsSrc.Cells(r, colNombre).Text)) Else nombre = ""
        If colUso > 0 Then usoCFDI = Trim$(CStr(wsSrc.Cells(r, colUso).Text)) Else usoCFDI = ""
        If colTipo > 0 Then tipoComp = Trim$(CStr(wsSrc.Cells(r, colTipo).Text)) Else tipoComp = ""
        If colForma > 0 Then formaPago = Trim$(CStr(wsSrc.Cells(r, colForma).Text)) Else formaPago = ""
        If colMetodo > 0 Then metodoPago = UCase$(Trim$(CStr(wsSrc.Cells(r, colMetodo).Text))) Else metodoPago = ""
        If colPDF > 0 Then pdfTxt = Trim$(CStr(wsSrc.Cells(r, colPDF).Text)) Else pdfTxt = ""
        If colConceptos > 0 Then conceptos = Trim$(CStr(wsSrc.Cells(r, colConceptos).Text)) Else conceptos = ""
        If colEstatus > 0 Then estatus = Trim$(CStr(wsSrc.Cells(r, colEstatus).Text)) Else estatus = ""

        If colSubTotal > 0 Then subTotal = C01X_ParseNumber(wsSrc.Cells(r, colSubTotal).Value) Else subTotal = 0
        If colISR > 0 Then isrVal = C01X_ParseNumber(wsSrc.Cells(r, colISR).Value) Else isrVal = 0
        If colIVA > 0 Then ivaVal = C01X_ParseNumber(wsSrc.Cells(r, colIVA).Value) Else ivaVal = 0
        If colIVA16 > 0 Then iva16Val = C01X_ParseNumber(wsSrc.Cells(r, colIVA16).Value) Else iva16Val = 0
        If colTotal > 0 Then totalVal = C01X_ParseNumber(wsSrc.Cells(r, colTotal).Value) Else totalVal = 0

        ' Siempre a LIMPIO
        C01X_WriteRow wsL, outL, wsSrc.Cells(r, colFechaEmi), uuid, pdfTxt, rfc, nombre, usoCFDI, tipoComp, formaPago, metodoPago, subTotal, isrVal, ivaVal, iva16Val, totalVal, conceptos
        outL = outL + 1

        ' Solo vigentes a PUE/PPD
        If C01X_EsCancelado(estatus) Then GoTo NextR

        If metodoPago = "PUE" Then
            C01X_WriteRow wsPUE, outP, wsSrc.Cells(r, colFechaEmi), uuid, pdfTxt, rfc, nombre, usoCFDI, tipoComp, formaPago, metodoPago, subTotal, isrVal, ivaVal, iva16Val, totalVal, conceptos
            outP = outP + 1
        ElseIf metodoPago = "PPD" Then
            C01X_WriteRow wsPPD, outD, wsSrc.Cells(r, colFechaEmi), uuid, pdfTxt, rfc, nombre, usoCFDI, tipoComp, formaPago, metodoPago, subTotal, isrVal, ivaVal, iva16Val, totalVal, conceptos
            outD = outD + 1
        End If

NextR:
    Next r

    C01X_ApplyCuentaMap wsL, mapL
    C01X_ApplyCuentaMap wsPUE, mapPUE
    C01X_ApplyCuentaMap wsPPD, mapPPD

    C01X_FormatOutput wsL
    C01X_FormatOutput wsPUE
    C01X_FormatOutput wsPPD

    C01X_SortByFecha wsL
    C01X_SortByFecha wsPUE
    C01X_SortByFecha wsPPD

End Sub


Private Sub C01X_WriteHeaders(ByVal ws As Worksheet)
    ws.Cells(1, 1).Value = "UUID"
    ws.Cells(1, 2).Value = "Fecha Emisión"
    ws.Cells(1, 3).Value = "PDF"
    ws.Cells(1, 4).Value = "RFC"
    ws.Cells(1, 5).Value = "Nombre, denominación o razón social del Receptor"
    ws.Cells(1, 6).Value = "Uso CFDI Receptor"
    ws.Cells(1, 7).Value = "Tipo.Com"
    ws.Cells(1, 8).Value = "Forma de Pago"
    ws.Cells(1, 9).Value = "Método de Pago"
    ws.Cells(1, 10).Value = "SubTotal XML"
    ws.Cells(1, 11).Value = "ISR"
    ws.Cells(1, 12).Value = "IVA"
    ws.Cells(1, 13).Value = "IVA 16"
    ws.Cells(1, 14).Value = "Total"
    ws.Cells(1, 15).Value = "Conceptos"
    ws.Cells(1, 16).Value = "Cuenta bancaria"
End Sub


Private Sub C01X_WriteRow( _
    ByVal ws As Worksheet, _
    ByVal targetRow As Long, _
    ByVal fechaCell As Range, _
    ByVal uuid As String, _
    ByVal pdfTxt As String, _
    ByVal rfc As String, _
    ByVal nombre As String, _
    ByVal usoCFDI As String, _
    ByVal tipoComp As String, _
    ByVal formaPago As String, _
    ByVal metodoPago As String, _
    ByVal subTotal As Double, _
    ByVal isrVal As Double, _
    ByVal ivaVal As Double, _
    ByVal iva16Val As Double, _
    ByVal totalVal As Double, _
    ByVal conceptos As String)

    Dim dt As Date

    ws.Cells(targetRow, 1).Value = uuid

    If C01X_TryParseCFDIFecha(fechaCell, dt) Then
        ws.Cells(targetRow, 2).Value = dt
    Else
        ws.Cells(targetRow, 2).Value = fechaCell.Text
    End If

    ws.Cells(targetRow, 3).Value = pdfTxt
    ws.Cells(targetRow, 4).Value = rfc
    ws.Cells(targetRow, 5).Value = nombre
    ws.Cells(targetRow, 6).Value = usoCFDI
    ws.Cells(targetRow, 7).Value = tipoComp
    ws.Cells(targetRow, 8).Value = formaPago
    ws.Cells(targetRow, 9).Value = metodoPago
    ws.Cells(targetRow, 10).Value = subTotal
    ws.Cells(targetRow, 11).Value = isrVal
    ws.Cells(targetRow, 12).Value = ivaVal
    ws.Cells(targetRow, 13).Value = iva16Val
    ws.Cells(targetRow, 14).Value = totalVal
    ws.Cells(targetRow, 15).Value = conceptos
End Sub


Private Sub C01X_FormatOutput(ByVal ws As Worksheet)

    Dim lastRow As Long
    lastRow = C01X_LastUsedRow(ws)

    ws.Columns(1).NumberFormat = "@"
    ws.Columns(3).NumberFormat = "@"
    ws.Columns(4).NumberFormat = "@"
    ws.Columns(16).NumberFormat = "@"

    If lastRow >= 2 Then
        ws.Range(ws.Cells(2, 2), ws.Cells(lastRow, 2)).NumberFormat = "dd/mm/yyyy"
        ws.Range(ws.Cells(2, 10), ws.Cells(lastRow, 14)).NumberFormat = "#,##0.00"
    End If

    ws.Rows(1).Font.Bold = True

    If ws.AutoFilterMode Then ws.AutoFilterMode = False
    ws.Range(ws.Cells(1, 1), ws.Cells(Application.Max(lastRow, 2), 16)).AutoFilter

    ws.Columns("A:P").EntireColumn.AutoFit
End Sub


Private Sub C01X_SortByFecha(ByVal ws As Worksheet)

    Dim lastRow As Long
    lastRow = C01X_LastUsedRow(ws)

    If lastRow < 3 Then Exit Sub

    ws.Range("A1:P" & lastRow).Sort _
        Key1:=ws.Range("B2:B" & lastRow), _
        Order1:=xlAscending, _
        Header:=xlYes
End Sub


Private Function C01X_EsCancelado(ByVal estatus As String) As Boolean
    Dim s As String
    s = UCase$(Trim$(estatus))

    If InStr(1, s, "CANCEL", vbTextCompare) > 0 Then
        C01X_EsCancelado = True
    Else
        C01X_EsCancelado = False
    End If
End Function


Private Function C01X_TryParseCFDIFecha(ByVal cel As Range, ByRef outDate As Date) As Boolean

    Dim s As String
    Dim soloFecha As String
    Dim p() As String
    Dim yy As Long, mm As Long, dd As Long
    Dim dt As Date

    On Error GoTo ErrHandler
    C01X_TryParseCFDIFecha = False

    If VarType(cel.Value) = vbDate Then
        outDate = CDate(cel.Value)
        C01X_TryParseCFDIFecha = True
        Exit Function
    End If

    s = Trim$(CStr(cel.Text))
    If s = "" Then Exit Function

    s = Replace(s, "T", " ")
    soloFecha = Split(s, " ")(0)
    soloFecha = Replace(soloFecha, ".", "-")
    soloFecha = Replace(soloFecha, "/", "-")

    p = Split(soloFecha, "-")
    If UBound(p) <> 2 Then Exit Function

    If Len(p(0)) = 4 Then
        yy = Val(p(0))
        mm = Val(p(1))
        dd = Val(p(2))
    Else
        dd = Val(p(0))
        mm = Val(p(1))
        yy = Val(p(2))
        If yy < 100 Then yy = 2000 + yy
    End If

    dt = DateSerial(yy, mm, dd)

    If Year(dt) <> yy Or Month(dt) <> mm Or Day(dt) <> dd Then Exit Function

    outDate = dt
    C01X_TryParseCFDIFecha = True
    Exit Function

ErrHandler:
    C01X_TryParseCFDIFecha = False
End Function


Private Function C01X_ParseNumber(ByVal v As Variant) As Double
    Dim s As String
    Dim neg As Boolean

    On Error GoTo ErrHandler

    If IsNumeric(v) Then
        C01X_ParseNumber = CDbl(v)
        Exit Function
    End If

    s = Trim$(CStr(v))
    If s = "" Then Exit Function

    s = Replace(s, Chr(160), "")
    s = Replace(s, "$", "")
    s = Replace(s, ",", "")
    s = Replace(s, " ", "")

    If Left$(s, 1) = "(" And Right$(s, 1) = ")" Then
        neg = True
        s = Replace(s, "(", "")
        s = Replace(s, ")", "")
    End If

    If IsNumeric(s) Then
        C01X_ParseNumber = CDbl(s)
        If neg Then C01X_ParseNumber = -C01X_ParseNumber
    End If
    Exit Function

ErrHandler:
    C01X_ParseNumber = 0
End Function


Private Function C01X_FindHeaderRowCFDI(ByVal ws As Worksheet) As Long
    Dim r As Long

    For r = 1 To 10
        If C01X_FindHeaderCol(ws, r, Array("UUID")) > 0 And _
           C01X_FindHeaderCol(ws, r, Array("Fecha Emisión", "Fecha Emision")) > 0 Then
            C01X_FindHeaderRowCFDI = r
            Exit Function
        End If
    Next r
End Function


Private Function C01X_FindHeaderCol(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headerNames As Variant) As Long
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim cellText As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        cellText = C01X_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headerNames) To UBound(headerNames)
            If cellText = C01X_NormHeader(CStr(headerNames(i))) Then
                C01X_FindHeaderCol = c
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function C01X_NormHeader(ByVal s As String) As String
    s = UCase$(Trim$(s))
    s = Replace(s, "Á", "A")
    s = Replace(s, "É", "E")
    s = Replace(s, "Í", "I")
    s = Replace(s, "Ó", "O")
    s = Replace(s, "Ú", "U")
    s = Replace(s, "Ü", "U")
    s = Replace(s, ".", "")
    s = Replace(s, ",", "")
    Do While InStr(s, "  ") > 0
        s = Replace(s, "  ", " ")
    Loop
    C01X_NormHeader = s
End Function


Private Function C01X_LastUsedRow(ByVal ws As Worksheet) As Long
    Dim c As Range

    Set c = ws.Cells.Find(What:="*", _
                          After:=ws.Range("A1"), _
                          LookIn:=xlFormulas, _
                          LookAt:=xlPart, _
                          SearchOrder:=xlByRows, _
                          SearchDirection:=xlPrevious, _
                          MatchCase:=False)

    If c Is Nothing Then
        C01X_LastUsedRow = 0
    Else
        C01X_LastUsedRow = c.Row
    End If
End Function


Private Function C01X_LastUsedCol(ByVal ws As Worksheet, ByVal hdrRow As Long) As Long
    Dim c As Range

    Set c = ws.Rows(hdrRow).Find(What:="*", _
                                 After:=ws.Cells(hdrRow, 1), _
                                 LookIn:=xlFormulas, _
                                 LookAt:=xlPart, _
                                 SearchOrder:=xlByColumns, _
                                 SearchDirection:=xlPrevious, _
                                 MatchCase:=False)

    If c Is Nothing Then
        C01X_LastUsedCol = 0
    Else
        C01X_LastUsedCol = c.Column
    End If
End Function


Private Function C01X_RowIsEmpty(ByVal ws As Worksheet, ByVal rowNum As Long, ByVal lastCol As Long) As Boolean
    Dim c As Long
    For c = 1 To lastCol
        If Trim$(CStr(ws.Cells(rowNum, c).Text)) <> "" Then Exit Function
    Next c
    C01X_RowIsEmpty = True
End Function


Private Function C01X_GetSheet(ByVal sheetName As String) As Worksheet
    On Error Resume Next
    Set C01X_GetSheet = ThisWorkbook.Worksheets(sheetName)
    On Error GoTo 0
End Function


Private Sub C01X_DeleteSheetIfExists(ByVal sheetName As String)
    Dim ws As Worksheet

    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(sheetName)
    On Error GoTo 0

    If Not ws Is Nothing Then
        Application.DisplayAlerts = False
        ws.Delete
        Application.DisplayAlerts = True
    End If
End Sub


Private Function C01X_LoadCuentaMap(ByVal sheetName As String) As Object

    Dim dict As Object
    Dim ws As Worksheet
    Dim colUUID As Long, colCuenta As Long, lastRow As Long
    Dim r As Long
    Dim uuid As String, cuenta As String

    Set dict = CreateObject("Scripting.Dictionary")

    Set ws = C01X_GetSheet(sheetName)
    If ws Is Nothing Then
        Set C01X_LoadCuentaMap = dict
        Exit Function
    End If

    colUUID = C01X_FindHeaderCol(ws, 1, Array("UUID"))
    colCuenta = C01X_FindHeaderCol(ws, 1, Array("Cuenta bancaria"))

    If colUUID = 0 Or colCuenta = 0 Then
        Set C01X_LoadCuentaMap = dict
        Exit Function
    End If

    lastRow = C01X_LastUsedRow(ws)
    If lastRow < 2 Then
        Set C01X_LoadCuentaMap = dict
        Exit Function
    End If

    For r = 2 To lastRow
        uuid = Trim$(CStr(ws.Cells(r, colUUID).Text))
        cuenta = Trim$(CStr(ws.Cells(r, colCuenta).Text))

        If uuid <> "" And cuenta <> "" Then
            If Not dict.Exists(uuid) Then dict.Add uuid, cuenta
        End If
    Next r

    Set C01X_LoadCuentaMap = dict
End Function


Private Sub C01X_ApplyCuentaMap(ByVal ws As Worksheet, ByVal dict As Object)

    Dim colUUID As Long, colCuenta As Long, lastRow As Long
    Dim r As Long
    Dim uuid As String

    If ws Is Nothing Then Exit Sub
    If dict Is Nothing Then Exit Sub

    colUUID = C01X_FindHeaderCol(ws, 1, Array("UUID"))
    colCuenta = C01X_FindHeaderCol(ws, 1, Array("Cuenta bancaria"))

    If colUUID = 0 Or colCuenta = 0 Then Exit Sub

    lastRow = C01X_LastUsedRow(ws)
    If lastRow < 2 Then Exit Sub

    For r = 2 To lastRow
        uuid = Trim$(CStr(ws.Cells(r, colUUID).Text))
        If uuid <> "" Then
            If dict.Exists(uuid) Then
                If Trim$(CStr(ws.Cells(r, colCuenta).Text)) = "" Then
                    ws.Cells(r, colCuenta).Value = dict(uuid)
                End If
            End If
        End If
    Next r
End Sub


Private Sub C01X_MergeMaps(ByVal targetDict As Object, ByVal sourceDict As Object)
    Dim k As Variant

    If targetDict Is Nothing Then Exit Sub
    If sourceDict Is Nothing Then Exit Sub

    For Each k In sourceDict.Keys
        If Not targetDict.Exists(CStr(k)) Then
            targetDict.Add CStr(k), sourceDict(k)
        End If
    Next k
End Sub
