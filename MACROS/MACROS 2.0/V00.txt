' Variable global para controlar la ejecución silenciosa de los mensajes en el macro maestro
Public g_SilentMode As Boolean

Public Sub V00_EJECUTAR_TODOS_LOS_REPORTES()
    ' Este macro maestro ejecuta todas las validaciones y reportes de forma secuencial
    
    On Error GoTo ErrHandler
    
    ' Activamos el modo silencioso para omitir los MsgBox intermedios
    g_SilentMode = True
    
    Call V01_VALIDACION_PREVIA
    Call V02_REPORTE_CONTROL_CARGOS_Y_FISCALES
    Call V03_VERIFICAR_VENTAS_BBVA_VS_BANCOS

    Call V08_RESUMEN_GENERAL_DE_CONCILIACION

    g_SilentMode = False

    Exit Sub

    g_SilentMode = False

Public Sub V01_VALIDACION_PREVIA()

    Dim outRow As Long

    Dim prevCalc As XlCalculation
    Dim prevScreen As Boolean

    On Error GoTo ErrHandler

    prevScreen = Application.ScreenUpdating

Private Function V01_ValidarHoja( _
    ByVal wsOut As Worksheet, _
    ByVal wsSal As Worksheet, _
    ByRef outRow As Long, _
    ByVal sheetName As String, _
    ByVal debeSer99 As Boolean, _
    ByVal validarS01 As Boolean, _
    ByVal validarTotal01 As Boolean) As Long

    Dim ws As Worksheet
    Dim hdrRow As Long
    Dim colUUID As Long, colForma As Long, colUso As Long, colTotal As Long
    Dim lastRow As Long
    Dim r As Long

    Dim uuidTxt As String
    Dim formaTxt As String
    Dim formaCod As String
    Dim usoTxt As String
    Dim usoCod As String
    Dim totalVal As Double
    Dim regla As String

    Set ws = V01_GetSheet(sheetName)

    If ws Is Nothing Then
        V01_AgregarHallazgo wsOut, wsSal, outRow, "V01_VALIDACION_PREVIA", sheetName, "", "", "ESTRUCTURA", "", "", "", "HOJA NO EXISTE"
        V01_ValidarHoja = 1
        Exit Function
    End If

    hdrRow = V01_FindHeaderRow(ws)
    If hdrRow = 0 Then
        V01_AgregarHallazgo wsOut, wsSal, outRow, "V01_VALIDACION_PREVIA", sheetName, "", "", "ESTRUCTURA", "", "", "", "NO SE IDENTIFICÓ FILA DE ENCABEZADOS"
        V01_ValidarHoja = 1
        Exit Function
    End If

    colUUID = V01_FindHeaderCol(ws, hdrRow, Array("UUID"))
    colForma = V01_FindHeaderCol(ws, hdrRow, Array("Forma de Pago", "Forma Pago"))

    If colUUID = 0 Or colForma = 0 Then
        V01_AgregarHallazgo wsOut, wsSal, outRow, "V01_VALIDACION_PREVIA", sheetName, "", "", "ESTRUCTURA", "", "", "", "FALTAN COLUMNAS REQUERIDAS (UUID / Forma de Pago)"
        V01_ValidarHoja = 1
        Exit Function
    End If

    If validarS01 Then
        colUso = V01_FindHeaderCol(ws, hdrRow, Array("Uso CFDI Receptor", "Uso CFDI", "UsoCFDI", "Uso CFDI del Receptor"))
        If colUso = 0 Then
            V01_AgregarHallazgo wsOut, wsSal, outRow, "V01_VALIDACION_PREVIA", sheetName, "", "", "ESTRUCTURA", "", "", "", "FALTA COLUMNA DE USO CFDI"
            V01_ValidarHoja = V01_ValidarHoja + 1
        End If
    End If

    If validarTotal01 Then
        colTotal = V01_FindHeaderCol(ws, hdrRow, Array("Total"))
        If colTotal = 0 Then
            V01_AgregarHallazgo wsOut, wsSal, outRow, "V01_VALIDACION_PREVIA", sheetName, "", "", "ESTRUCTURA", "", "", "", "FALTA COLUMNA TOTAL"
            V01_ValidarHoja = V01_ValidarHoja + 1
        End If
    End If

    lastRow = V01_LastUsedRow(ws)
    If lastRow <= hdrRow Then Exit Function

    For r = hdrRow + 1 To lastRow

        uuidTxt = Trim$(CStr(ws.Cells(r, colUUID).Value))
        formaTxt = Trim$(CStr(ws.Cells(r, colForma).Value))

        If uuidTxt = "" And formaTxt = "" Then GoTo SiguienteR

        formaCod = V01_ExtraerCodigoFormaPago(formaTxt)

        ' Regla Forma de Pago 99 / no 99
        If debeSer99 Then
            If formaCod <> "99" Then
                regla = "Forma de Pago debe ser 99"
                V01_AgregarHallazgo wsOut, wsSal, outRow, "V01_VALIDACION_PREVIA", sheetName, r, uuidTxt, "Forma de Pago", formaTxt, formaCod, regla, "EN PPD SOLO SE ACEPTA 99"
                V01_ValidarHoja = V01_ValidarHoja + 1
            End If
        Else
            If formaCod = "99" Then
                regla = "Forma de Pago no debe ser 99"
                V01_AgregarHallazgo wsOut, wsSal, outRow, "V01_VALIDACION_PREVIA", sheetName, r, uuidTxt, "Forma de Pago", formaTxt, formaCod, regla, "EN PUE NO DEBE EXISTIR 99"
                V01_ValidarHoja = V01_ValidarHoja + 1
            End If
        End If

        ' Regla CFDI_E_PUE: Forma de Pago 01 -> Total < 2000
        If validarTotal01 And colTotal > 0 Then
            If formaCod = "01" Then
                totalVal = V01_ParseNumber(ws.Cells(r, colTotal).Value)
                If totalVal >= 2000 Then
                    regla = "Si Forma de Pago = 01, Total debe ser menor a 2000"
                    V01_AgregarHallazgo wsOut, wsSal, outRow, "V01_VALIDACION_PREVIA", sheetName, r, uuidTxt, "Total", CStr(ws.Cells(r, colTotal).Text), CStr(totalVal), regla, "FORMA DE PAGO 01 CON TOTAL IGUAL O MAYOR A 2000"
                    V01_ValidarHoja = V01_ValidarHoja + 1
                End If
            End If
        End If

        ' Regla Uso CFDI: no S01 (solo CFDI_E_PUE y CFDI_E_PPD)
        If validarS01 And colUso > 0 Then
            usoTxt = Trim$(CStr(ws.Cells(r, colUso).Value))
            usoCod = V01_ExtraerUsoCFDI(usoTxt)

            If usoCod = "S01" Then
                regla = "Uso CFDI no debe ser S01"
                V01_AgregarHallazgo wsOut, wsSal, outRow, "V01_VALIDACION_PREVIA", sheetName, r, uuidTxt, "Uso CFDI", usoTxt, usoCod, regla, "NO DEBE EXISTIR S01 EN ESTE RUBRO"
                V01_ValidarHoja = V01_ValidarHoja + 1
            End If
        End If

SiguienteR:
    Next r
End Function

Private Function V01_ExtraerCodigoFormaPago(ByVal s As String) As String
    Dim txt As String
    Dim i As Long, ch As String
    Dim digitos As String

    txt = Trim$(s)
    If txt = "" Then Exit Function

    For i = 1 To Len(txt)
        ch = Mid$(txt, i, 1)
        If ch Like "#" Then
            digitos = digitos & ch
            If Len(digitos) = 2 Then Exit For
        ElseIf Len(digitos) > 0 Then
            Exit For
        End If
    Next i

    If Len(digitos) = 1 Then digitos = "0" & digitos
    If Len(digitos) >= 2 Then V01_ExtraerCodigoFormaPago = Left$(digitos, 2)
End Function


Private Function V01_ExtraerUsoCFDI(ByVal s As String) As String
    Dim txt As String

    txt = UCase$(Trim$(s))
    txt = Replace(txt, Chr(160), " ")

    If InStr(1, txt, "S01", vbTextCompare) > 0 Then
        V01_ExtraerUsoCFDI = "S01"
        Exit Function
    End If

    If Len(txt) >= 3 Then
        V01_ExtraerUsoCFDI = Left$(txt, 3)
    End If
End Function


Private Function V01_ParseNumber(ByVal v As Variant) As Double
    Dim s As String
    Dim neg As Boolean

    On Error GoTo ErrHandler

    If IsNumeric(v) Then
        V01_ParseNumber = CDbl(v)
        Exit Function
    End If

    s = Trim$(CStr(v))
    If s = "" Then Exit Function

    s = Replace(s, Chr(160), "")
    s = Replace(s, "$", "")
    s = Replace(s, ",", "")
    s = Replace(s, " ", "")

    If Left$(s, 1) = "(" And Right$(s, 1) = ")" Then
        neg = True
        s = Replace(s, "(", "")
        s = Replace(s, ")", "")
    End If

    If IsNumeric(s) Then
        V01_ParseNumber = CDbl(s)
        If neg Then V01_ParseNumber = -V01_ParseNumber
    End If
    Exit Function

ErrHandler:
    V01_ParseNumber = 0
End Function

Private Function V01_FindHeaderRow(ByVal ws As Worksheet) As Long
    Dim r As Long

    For r = 1 To 10
        If V01_FindHeaderCol(ws, r, Array("UUID")) > 0 And _
           V01_FindHeaderCol(ws, r, Array("Forma de Pago", "Forma Pago")) > 0 Then
            V01_FindHeaderRow = r
            Exit Function
        End If
    Next r
End Function


Private Function V01_FindHeaderCol(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headerNames As Variant) As Long
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim cellText As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        cellText = V01_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headerNames) To UBound(headerNames)
            If cellText = V01_NormHeader(CStr(headerNames(i))) Then
                V01_FindHeaderCol = c
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function V01_NormHeader(ByVal s As String) As String
    s = UCase$(Trim$(s))
    s = Replace(s, "Á", "A")
    s = Replace(s, "É", "E")
    s = Replace(s, "Í", "I")
    s = Replace(s, "Ó", "O")
    s = Replace(s, "Ú", "U")
    s = Replace(s, "Ü", "U")
    s = Replace(s, ".", "")
    s = Replace(s, ",", "")
    Do While InStr(s, "  ") > 0
        s = Replace(s, "  ", " ")
    Loop
    V01_NormHeader = s
End Function


Private Function V01_LastUsedRow(ByVal ws As Worksheet) As Long
    Dim c As Range

    Set c = ws.Cells.Find(What:="*", _
                          After:=ws.Range("A1"), _
                          LookIn:=xlFormulas, _
                          LookAt:=xlPart, _
                          SearchOrder:=xlByRows, _
                          SearchDirection:=xlPrevious, _
                          MatchCase:=False)

    If c Is Nothing Then
        V01_LastUsedRow = 0
    Else
        V01_LastUsedRow = c.Row
    End If
End Function


Private Sub V01_DeleteSheetIfExists(ByVal sheetName As String)
    Dim ws As Worksheet

    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(sheetName)
    On Error GoTo 0

    If Not ws Is Nothing Then
        Application.DisplayAlerts = False
        ws.Delete
        Application.DisplayAlerts = True
    End If
End Sub

        "CFDI_E_PUE", _
        pueTotal, _
        pueConCta, _
        pueSinCta, _
        V02_Pct(pueConCta, pueTotal), _
        obsPUE)

    wsOut.Range("A" & outRow + 4 & ":F" & outRow + 4).Value = Array( _
        IIf(wsPagos Is Nothing, "PAGOS E / COMPL. E", wsPagos.Name), _
        pagosTotal, _
        pagosConCta, _
        pagosSinCta, _
        V02_Pct(pagosConCta, pagosTotal), _
        obsPagos)

    wsOut.Range("E" & outRow + 3 & ":E" & outRow + 4).NumberFormat = "0.00%"

    outRow = outRow + 7

    ' =====================================================
    ' TABLA 4 - DETALLE DE PENDIENTES REALES
    ' =====================================================

            End If
        End If
    Next ws

    If Not wsEST Is Nothing Then

    End If

    If outRow = detallePendStart + 3 Then

    ' =====================================================

    wsOut.Cells(detallePPDStart, 1).Value = "TABLA 5 - DETALLE DE PPD SIN UUID DE PAGO"

        If V02_EsHojaNumerica(ws.Name) Then

        outRow = outRow + 1
    End If

    wsOut.Columns("A:J").EntireColumn.AutoFit

    Application.Calculation = prevCalc
    Application.EnableEvents = prevEvents
    Application.ScreenUpdating = prevScreen

    If Not g_SilentMode Then MsgBox "Reporte V02 generado correctamente.", vbInformation
    Exit Sub

ErrHandler:

    Application.EnableEvents = prevEvents
    Application.ScreenUpdating = prevScreen

Private Sub V02_ContarDetalleBanco( _
    ByVal ws As Worksheet, _
    ByRef total As Long, _
    ByRef conObs As Long, _
    ByRef pue As Long, _
    ByRef ppd As Long, _
    ByRef ppdSinUUIDPago As Long, _
    ByRef preclasificados As Long, _
    ByRef pendientes As Long)

    Dim hdrDia As Range, hdrCargo As Range, hdrAbono As Range
    Dim hdrRow As Long
    Dim colCargo As Long, colAbono As Long, colCompl As Long, colObs As Long
    Dim lastRow As Long, r As Long
    Dim cargoVal As Double, abonoVal As Double
    Dim complTxt As String, obsTxt As String

    Set hdrDia = ws.Cells.Find(What:="Día", LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    If hdrDia Is Nothing Then
        Set hdrDia = ws.Cells.Find(What:="Dia", LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    End If
    If hdrDia Is Nothing Then Exit Sub

    hdrRow = hdrDia.Row

    Set hdrCargo = V02_FindHeader(ws, hdrRow, Array("Cargo"))
    Set hdrAbono = V02_FindHeader(ws, hdrRow, Array("Abono"))

    If hdrCargo Is Nothing Or hdrAbono Is Nothing Then Exit Sub

    colCargo = hdrCargo.Column
    colAbono = hdrAbono.Column
    colCompl = V02_FindHeaderCol(ws, hdrRow, Array("COMPL. CTA"))
    colObs = V02_FindHeaderCol(ws, hdrRow, Array("OBS"))

    lastRow = V02_LastUsedRow(ws)
    If lastRow <= hdrRow Then Exit Sub

    For r = hdrRow + 1 To lastRow
        cargoVal = V02_ParseNumber(ws.Cells(r, colCargo).Value)
        abonoVal = V02_ParseNumber(ws.Cells(r, colAbono).Value)

        If cargoVal > 0 And abonoVal = 0 Then
            total = total + 1

            obsTxt = IIf(colObs > 0, Trim$(CStr(ws.Cells(r, colObs).Value)), "")
            complTxt = IIf(colCompl > 0, UCase$(Trim$(CStr(ws.Cells(r, colCompl).Value))), "")

            If obsTxt <> "" Then conObs = conObs + 1
            If complTxt = "PUE" Then pue = pue + 1
            If complTxt = "PPD" Then
                ppd = ppd + 1
                If obsTxt = "" Then ppdSinUUIDPago = ppdSinUUIDPago + 1
            End If

            If obsTxt <> "" And complTxt <> "PUE" And complTxt <> "PPD" Then
                preclasificados = preclasificados + 1
            End If

            If obsTxt = "" And complTxt = "" Then
                pendientes = pendientes + 1
            End If
        End If
    Next r
End Sub

Private Sub V02_ContarPPDvsComplemento( _
    ByVal wsPPD As Worksheet, _
    ByVal wsPagos As Worksheet, _
    ByRef totalPPD As Long, _
    ByRef conComp As Long, _
    ByRef sinComp As Long, _
    ByRef observacion As String)

    Dim dictPagos As Object
    Dim hdrPPD As Long, lastPPD As Long
    Dim hdrPagos As Long, lastPagos As Long
    Dim colPPDUUID As Long
    Dim colPagUUID As Long, colPagMother As Long
    Dim r As Long, k As String

    Set dictPagos = CreateObject("Scripting.Dictionary")

    If wsPPD Is Nothing Then
        observacion = "No existe CFDI_E_PPD"
        Exit Sub
    End If

    hdrPPD = V02_FindHeaderRowSimple(wsPPD, Array("UUID"))
    If hdrPPD = 0 Then
        observacion = "No se identificó encabezado en CFDI_E_PPD"
        Exit Sub
    End If

    colPPDUUID = V02_FindHeaderCol(wsPPD, hdrPPD, Array("UUID"))
    If colPPDUUID = 0 Then
        observacion = "No existe columna UUID en CFDI_E_PPD"
        Exit Sub
    End If

    If wsPagos Is Nothing Then
        lastPPD = V02_LastUsedRow(wsPPD)
        If lastPPD > hdrPPD Then
            For r = hdrPPD + 1 To lastPPD
                If Trim$(CStr(wsPPD.Cells(r, colPPDUUID).Value)) <> "" Then totalPPD = totalPPD + 1
            Next r
        End If
        sinComp = totalPPD
        observacion = "No existe PAGOS E / COMPL. E"
        Exit Sub
    End If

    hdrPagos = V02_FindHeaderRowSimple(wsPagos, Array("UUID", "UUID Madre", "UUID MADRE"))
    If hdrPagos > 0 Then
        colPagUUID = V02_FindHeaderCol(wsPagos, hdrPagos, Array("UUID"))
        colPagMother = V02_FindHeaderCol(wsPagos, hdrPagos, Array("UUID Madre", "UUID MADRE", "UUID_MADRE"))

        lastPagos = V02_LastUsedRow(wsPagos)
        For r = hdrPagos + 1 To lastPagos
            If colPagUUID > 0 Then
                k = V02_NormKey(wsPagos.Cells(r, colPagUUID).Value)
                If k <> "" Then If Not dictPagos.Exists(k) Then dictPagos.Add k, True
            End If

            If colPagMother > 0 Then
                k = V02_NormKey(wsPagos.Cells(r, colPagMother).Value)
                If k <> "" Then If Not dictPagos.Exists(k) Then dictPagos.Add k, True
            End If
        Next r
    End If

    lastPPD = V02_LastUsedRow(wsPPD)
    For r = hdrPPD + 1 To lastPPD
        k = V02_NormKey(wsPPD.Cells(r, colPPDUUID).Value)
        If k <> "" Then
            totalPPD = totalPPD + 1
            If dictPagos.Exists(k) Then
                conComp = conComp + 1
            Else
                sinComp = sinComp + 1
            End If
        End If
    Next r

    If observacion = "" Then observacion = "OK"
End Sub


Private Sub V02_ContarCuentaBancaria( _
    ByVal ws As Worksheet, _
    ByVal uuidHeader As String, _
    ByVal ctaHeader As String, _
    ByRef total As Long, _
    ByRef conCta As Long, _
    ByRef sinCta As Long, _
    ByRef observacion As String)

    Dim hdrRow As Long, lastRow As Long
    Dim colUUID As Long, colCta As Long
    Dim r As Long
    Dim uuidTxt As String, ctaTxt As String

    If ws Is Nothing Then
        observacion = "Hoja no existe"
        Exit Sub
    End If

    hdrRow = V02_FindHeaderRowSimple(ws, Array(uuidHeader))
    If hdrRow = 0 Then
        observacion = "No se identificó encabezado"
        Exit Sub
    End If

    colUUID = V02_FindHeaderCol(ws, hdrRow, Array(uuidHeader))
    colCta = V02_FindHeaderCol(ws, hdrRow, Array(ctaHeader))

    If colUUID = 0 Or colCta = 0 Then
        observacion = "Faltan columnas UUID / Cuenta bancaria"
        Exit Sub
    End If

    lastRow = V02_LastUsedRow(ws)
    If lastRow <= hdrRow Then
        observacion = "Sin datos"
        Exit Sub
    End If

    For r = hdrRow + 1 To lastRow
        uuidTxt = Trim$(CStr(ws.Cells(r, colUUID).Value))
        ctaTxt = Trim$(CStr(ws.Cells(r, colCta).Value))

        If uuidTxt <> "" Then
            total = total + 1
            If ctaTxt <> "" Then
                conCta = conCta + 1
            Else
                sinCta = sinCta + 1
            End If
        End If
    Next r

    If observacion = "" Then observacion = "OK"
End Sub

    Dim arr, i As Long

Private Function V02_FindHeader(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headers As Variant) As Range
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim txt As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        txt = V02_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headers) To UBound(headers)
            If txt = V02_NormHeader(CStr(headers(i))) Then
                Set V02_FindHeader = ws.Cells(hdrRow, c)
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function V02_FindHeaderCol(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headerNames As Variant) As Long
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim cellText As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        cellText = V02_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headerNames) To UBound(headerNames)
            If cellText = V02_NormHeader(CStr(headerNames(i))) Then
                V02_FindHeaderCol = c
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function V02_NormHeader(ByVal s As String) As String
    s = UCase$(Trim$(s))
    s = Replace(s, "Á", "A")
    s = Replace(s, "É", "E")
    s = Replace(s, "Í", "I")
    s = Replace(s, "Ó", "O")
    s = Replace(s, "Ú", "U")
    s = Replace(s, "Ü", "U")
    s = Replace(s, ".", "")
    s = Replace(s, ",", "")
    s = Replace(s, "/", "")
    Do While InStr(s, "  ") > 0
        s = Replace(s, "  ", " ")
    Loop
    V02_NormHeader = s
End Function


Private Function V02_LastUsedRow(ByVal ws As Worksheet) As Long
    Dim c As Range

    Set c = ws.Cells.Find(What:="*", _
                          After:=ws.Range("A1"), _
                          LookIn:=xlFormulas, _
                          LookAt:=xlPart, _
                          SearchOrder:=xlByRows, _
                          SearchDirection:=xlPrevious, _
                          MatchCase:=False)

    If c Is Nothing Then
        V02_LastUsedRow = 0
    Else
        V02_LastUsedRow = c.Row
    End If
End Function


Private Function V02_ParseNumber(ByVal v As Variant) As Double
    Dim s As String
    Dim neg As Boolean

    On Error GoTo ErrHandler

    If IsNumeric(v) Then
        V02_ParseNumber = CDbl(v)
        Exit Function
    End If

    s = Trim$(CStr(v))
    If s = "" Then Exit Function

    s = Replace(s, Chr(160), "")
    s = Replace(s, "$", "")
    s = Replace(s, ",", "")
    s = Replace(s, " ", "")

    If Left$(s, 1) = "(" And Right$(s, 1) = ")" Then
        neg = True
        s = Replace(s, "(", "")
        s = Replace(s, ")", "")
    End If

    If IsNumeric(s) Then
        V02_ParseNumber = CDbl(s)
        If neg Then V02_ParseNumber = -V02_ParseNumber
    End If
    Exit Function

ErrHandler:
    V02_ParseNumber = 0
End Function

End Function

End Function

    Dim s As String

    s = Trim$(CStr(v))
    If s = "" Then Exit Function

    s = UCase$(s)
    s = Replace(s, Chr(160), "")
    s = Replace(s, " ", "")
    s = Replace(s, "-", "")
    s = Replace(s, "_", "")
    s = Replace(s, ".", "")

    V02_NormKey = s
End Function

        V02_Pct = 0

End Function

Private Sub V03_ContarVentasAgrupadas( _
    ByVal ws As Worksheet, _
    ByRef total As Long, _
    ByRef conCuenta As Long, _
    ByRef sinCuenta As Long)

    Dim hdrRow As Long
    Dim colId As Long, colCuenta As Long
    Dim lastRow As Long, r As Long
    Dim dictCuenta As Object
    Dim idv As String, cuenta As String
    Dim k As Variant

    If ws Is Nothing Then Exit Sub

    hdrRow = V03_FindHeaderRowVentas(ws)
    If hdrRow = 0 Then Exit Sub

    colId = V03_FindHeaderCol(ws, hdrRow, Array("ID_venta", "id_venta"))
    colCuenta = V03_FindHeaderCol(ws, hdrRow, Array("Cuenta bancaria"))

    If colId = 0 Then Exit Sub

    Set dictCuenta = CreateObject("Scripting.Dictionary")

    lastRow = V03_LastUsedRow(ws)
    If lastRow <= hdrRow Then Exit Sub

    For r = hdrRow + 1 To lastRow
        idv = Trim$(CStr(ws.Cells(r, colId).Value))
        If idv <> "" Then
            cuenta = V03_GetCellText(ws, r, colCuenta)

            If Not dictCuenta.Exists(idv) Then
                dictCuenta.Add idv, cuenta
            Else
                If Trim$(CStr(dictCuenta(idv))) = "" And cuenta <> "" Then
                    dictCuenta(idv) = cuenta
                End If
            End If
        End If
    Next r

    For Each k In dictCuenta.Keys
        total = total + 1
        If Trim$(CStr(dictCuenta(k))) <> "" Then
            conCuenta = conCuenta + 1
        Else
            sinCuenta = sinCuenta + 1
        End If
    Next k
End Sub


Private Sub V03_ContarAbonosBanco( _
    ByVal ws As Worksheet, _
    ByRef total As Long, _
    ByRef conId As Long, _
    ByRef sinId As Long)

    Dim hdrDia As Range, hdrCargo As Range, hdrAbono As Range
    Dim hdrRow As Long
    Dim colCargo As Long, colAbono As Long, colId As Long
    Dim lastRow As Long, r As Long
    Dim cargoVal As Double, abonoVal As Double
    Dim idTxt As String

    Set hdrDia = ws.Cells.Find(What:="Día", LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    If hdrDia Is Nothing Then
        Set hdrDia = ws.Cells.Find(What:="Dia", LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    End If
    If hdrDia Is Nothing Then Exit Sub

    hdrRow = hdrDia.Row

    Set hdrCargo = V03_FindHeader(ws, hdrRow, Array("Cargo"))
    Set hdrAbono = V03_FindHeader(ws, hdrRow, Array("Abono"))

    If hdrCargo Is Nothing Or hdrAbono Is Nothing Then Exit Sub

    colCargo = hdrCargo.Column
    colAbono = hdrAbono.Column
    colId = V03_FindHeaderCol(ws, hdrRow, Array("ID_venta", "id_venta"))

    lastRow = V03_LastUsedRow(ws)
    If lastRow <= hdrRow Then Exit Sub

    For r = hdrRow + 1 To lastRow
        cargoVal = V03_ParseNumber(ws.Cells(r, colCargo).Value)
        abonoVal = V03_ParseNumber(ws.Cells(r, colAbono).Value)

        If abonoVal > 0 And cargoVal = 0 Then
            total = total + 1
            idTxt = V03_GetCellText(ws, r, colId)

            If idTxt <> "" Then
                conId = conId + 1
            Else
                sinId = sinId + 1
            End If
        End If
    Next r
End Sub

Private Sub V03_AgregarDetalleVentasPendientes( _
    ByVal ws As Worksheet, _
    ByVal wsOut As Worksheet, _
    ByRef outRow As Long)

    Dim hdrRow As Long
    Dim colId As Long, colUUID As Long, colCuenta As Long
    Dim colTotal As Long, colImporte As Long, colMonto As Long, colPrecio As Long, colCantidad As Long
    Dim lastRow As Long, r As Long

    Dim dictUUID As Object, dictMonto As Object, dictCuenta As Object
    Dim idv As String, uuidTxt As String, cuentaTxt As String
    Dim montoLinea As Double
    Dim k As Variant

    If ws Is Nothing Then Exit Sub

    hdrRow = V03_FindHeaderRowVentas(ws)
    If hdrRow = 0 Then Exit Sub

    colId = V03_FindHeaderCol(ws, hdrRow, Array("ID_venta", "id_venta"))
    colUUID = V03_FindHeaderCol(ws, hdrRow, Array("UUID", "uuid"))
    colCuenta = V03_FindHeaderCol(ws, hdrRow, Array("Cuenta bancaria"))

    colTotal = V03_FindHeaderCol(ws, hdrRow, Array("Total", "total"))
    colImporte = V03_FindHeaderCol(ws, hdrRow, Array("Importe", "importe"))
    colMonto = V03_FindHeaderCol(ws, hdrRow, Array("Monto", "monto", "Monto total", "monto total", "Total venta", "total venta", "Valor de la operación", "Valor de la operacion"))
    colPrecio = V03_FindHeaderCol(ws, hdrRow, Array("precio_unitario", "precio unitario", "Precio_Unitario", "Precio Unitario"))
    colCantidad = V03_FindHeaderCol(ws, hdrRow, Array("Cantidad", "cantidad", "Qty", "qty", "Cant", "cant"))

    If colId = 0 Then Exit Sub

    Set dictUUID = CreateObject("Scripting.Dictionary")
    Set dictMonto = CreateObject("Scripting.Dictionary")
    Set dictCuenta = CreateObject("Scripting.Dictionary")

    lastRow = V03_LastUsedRow(ws)
    If lastRow <= hdrRow Then Exit Sub

    For r = hdrRow + 1 To lastRow
        idv = Trim$(CStr(ws.Cells(r, colId).Value))
        If idv <> "" Then
            uuidTxt = V03_GetCellText(ws, r, colUUID)
            cuentaTxt = V03_GetCellText(ws, r, colCuenta)
            montoLinea = V03_GetVentaLineAmount(ws, r, colTotal, colImporte, colMonto, colPrecio, colCantidad)

            If Not dictUUID.Exists(idv) Then
                dictUUID.Add idv, uuidTxt
                dictMonto.Add idv, CDbl(0)
                dictCuenta.Add idv, cuentaTxt
            End If

            dictMonto(idv) = CDbl(dictMonto(idv)) + montoLinea

            If Trim$(CStr(dictUUID(idv))) = "" And uuidTxt <> "" Then
                dictUUID(idv) = uuidTxt
            End If

            If Trim$(CStr(dictCuenta(idv))) = "" And cuentaTxt <> "" Then
                dictCuenta(idv) = cuentaTxt
            End If
        End If
    Next r

    For Each k In dictUUID.Keys
        If Trim$(CStr(dictCuenta(k))) = "" Then
            wsOut.Cells(outRow, 1).Value = k
            wsOut.Cells(outRow, 2).Value = dictUUID(k)
            wsOut.Cells(outRow, 3).Value = dictMonto(k)
            wsOut.Cells(outRow, 4).Value = dictCuenta(k)
            wsOut.Cells(outRow, 5).Value = "VENTA SIN CUENTA BANCARIA"
            outRow = outRow + 1
        End If
    Next k
End Sub

Private Function V03_GetVentaLineAmount( _
    ByVal ws As Worksheet, _
    ByVal rowNum As Long, _
    ByVal colVTotal As Long, _
    ByVal colVImporte As Long, _
    ByVal colVMonto As Long, _
    ByVal colVPrecio As Long, _
    ByVal colVCantidad As Long) As Double

    Dim totalVal As Double
    Dim importeVal As Double
    Dim montoVal As Double
    Dim precioVal As Double
    Dim cantidadVal As Double

    If colVTotal > 0 Then
        totalVal = V03_ParseNumber(ws.Cells(rowNum, colVTotal).Value)
        If totalVal <> 0 Then
            V03_GetVentaLineAmount = totalVal
            Exit Function
        End If
    End If

    If colVImporte > 0 Then
        importeVal = V03_ParseNumber(ws.Cells(rowNum, colVImporte).Value)
        If importeVal <> 0 Then
            V03_GetVentaLineAmount = importeVal
            Exit Function
        End If
    End If

    If colVMonto > 0 Then
        montoVal = V03_ParseNumber(ws.Cells(rowNum, colVMonto).Value)
        If montoVal <> 0 Then
            V03_GetVentaLineAmount = montoVal
            Exit Function
        End If
    End If

    If colVPrecio > 0 Then
        precioVal = V03_ParseNumber(ws.Cells(rowNum, colVPrecio).Value)

        If colVCantidad > 0 Then
            cantidadVal = V03_ParseNumber(ws.Cells(rowNum, colVCantidad).Value)

Private Function V03_NormHeader(ByVal s As String) As String

    s = Replace(s, ",", "")
    s = Replace(s, "/", "")

End Function

Private Function V03_ParseNumber(ByVal v As Variant) As Double
    Dim s As String
    Dim neg As Boolean

    On Error GoTo ErrHandler

    If IsNumeric(v) Then
        V03_ParseNumber = CDbl(v)
        Exit Function
    End If

    s = Trim$(CStr(v))
    If s = "" Then Exit Function

    s = Replace(s, Chr(160), "")
    s = Replace(s, "$", "")
    s = Replace(s, ",", "")
    s = Replace(s, " ", "")

    If Left$(s, 1) = "(" And Right$(s, 1) = ")" Then
        neg = True
        s = Replace(s, "(", "")
        s = Replace(s, ")", "")
    End If

    If IsNumeric(s) Then
        V03_ParseNumber = CDbl(s)
        If neg Then V03_ParseNumber = -V03_ParseNumber
    End If
    Exit Function

ErrHandler:
    V03_ParseNumber = 0
End Function

Private Function V04_IsOKMatch(ByVal okValue As String) As Boolean
    Dim txt As String
    txt = UCase$(Trim$(okValue))

    If txt = "" Then Exit Function

    If InStr(1, txt, "OK", vbTextCompare) > 0 Then
        V04_IsOKMatch = True
    End If
End Function


Private Function V04_BuildMotivoMatch( _
    ByVal okValue As String, _
    ByVal uuidTxt As String, _
    ByVal transTxt As String) As String

    Dim okTxt As String

    okTxt = Trim$(okValue)

    If okTxt = "" Then
        If uuidTxt = "" And transTxt = "" Then
            V04_BuildMotivoMatch = "SIN OK_MP / SIN UUID / SIN numero_transaccion"
        ElseIf uuidTxt = "" Then
            V04_BuildMotivoMatch = "SIN OK_MP / SIN UUID"
        ElseIf transTxt = "" Then
            V04_BuildMotivoMatch = "SIN OK_MP / SIN numero_transaccion"
        Else
            V04_BuildMotivoMatch = "SIN MATCH MP"
        End If
    Else
        If uuidTxt = "" And transTxt = "" Then
            V04_BuildMotivoMatch = "OK_MP SIN 'OK' / SIN UUID / SIN numero_transaccion"
        ElseIf uuidTxt = "" Then
            V04_BuildMotivoMatch = "OK_MP SIN 'OK' / SIN UUID"
        ElseIf transTxt = "" Then
            V04_BuildMotivoMatch = "OK_MP SIN 'OK' / SIN numero_transaccion"
        Else
            V04_BuildMotivoMatch = "OK_MP SIN 'OK'"
        End If
    End If
End Function


Private Function V04_BuildMotivoUUID( _
    ByVal okValue As String, _
    ByVal transTxt As String) As String

    If V04_IsOKMatch(okValue) Then
        If transTxt = "" Then
            V04_BuildMotivoUUID = "SIN UUID / CON MATCH MP / SIN numero_transaccion"
        Else
            V04_BuildMotivoUUID = "SIN UUID / CON MATCH MP"
        End If
    Else
        If transTxt = "" Then
            V04_BuildMotivoUUID = "SIN UUID / SIN MATCH MP / SIN numero_transaccion"
        Else
            V04_BuildMotivoUUID = "SIN UUID / SIN MATCH MP"
        End If
    End If
End Function

Private Sub V05_ContarVentasAgrupadas( _
    ByVal ws As Worksheet, _
    ByRef total As Long, _
    ByRef conCuenta As Long, _
    ByRef sinCuenta As Long)

    Dim dictTipo As Object, dictTrans As Object, dictIds As Object, dictUUIDs As Object, dictMonto As Object, dictCuenta As Object

    Set dictTipo = CreateObject("Scripting.Dictionary")
    Set dictTrans = CreateObject("Scripting.Dictionary")
    Set dictIds = CreateObject("Scripting.Dictionary")
    Set dictUUIDs = CreateObject("Scripting.Dictionary")
    Set dictMonto = CreateObject("Scripting.Dictionary")
    Set dictCuenta = CreateObject("Scripting.Dictionary")

    V05_BuildVentasGroupingData ws, dictTipo, dictTrans, dictIds, dictUUIDs, dictMonto, dictCuenta

    Dim k As Variant
    For Each k In dictTipo.Keys
        total = total + 1
        If Trim$(CStr(dictCuenta(k))) <> "" Then
            conCuenta = conCuenta + 1
        Else
            sinCuenta = sinCuenta + 1
        End If
    Next k
End Sub


Private Sub V05_BuildVentasGroupingData( _
    ByVal ws As Worksheet, _
    ByVal dictTipo As Object, _
    ByVal dictTrans As Object, _
    ByVal dictIds As Object, _
    ByVal dictUUIDs As Object, _
    ByVal dictMonto As Object, _
    ByVal dictCuenta As Object)

    Dim hdrRow As Long
    Dim colId As Long, colUUID As Long, colCuenta As Long, colTrans As Long
    Dim colTotal As Long, colImporte As Long, colMonto As Long, colPrecio As Long, colCantidad As Long
    Dim lastRow As Long, r As Long

    Dim idv As String, uuidTxt As String, cuentaTxt As String, transTxt As String
    Dim groupKey As String, tipoGrupo As String
    Dim montoLinea As Double

    If ws Is Nothing Then Exit Sub

    hdrRow = V05_FindHeaderRowVentas(ws)
    If hdrRow = 0 Then Exit Sub

    colId = V05_FindHeaderCol(ws, hdrRow, Array("ID_venta", "id_venta"))
    colUUID = V05_FindHeaderCol(ws, hdrRow, Array("UUID", "uuid"))
    colCuenta = V05_FindHeaderCol(ws, hdrRow, Array("Cuenta bancaria"))
    colTrans = V05_FindHeaderCol(ws, hdrRow, Array("numero_transaccion", "Número de transacción", "Numero de transaccion", "numero transaccion", "Numero transaccion", "Operacion relacionada"))

    colTotal = V05_FindHeaderCol(ws, hdrRow, Array("Total", "total"))
    colImporte = V05_FindHeaderCol(ws, hdrRow, Array("Importe", "importe"))
    colMonto = V05_FindHeaderCol(ws, hdrRow, Array("Monto", "monto", "Monto total", "monto total", "Total venta", "total venta", "Valor de la operación", "Valor de la operacion"))
    colPrecio = V05_FindHeaderCol(ws, hdrRow, Array("precio_unitario", "precio unitario", "Precio_Unitario", "Precio Unitario"))
    colCantidad = V05_FindHeaderCol(ws, hdrRow, Array("Cantidad", "cantidad", "Qty", "qty", "Cant", "cant"))

    If colId = 0 Then Exit Sub

    lastRow = V05_LastUsedRow(ws)
    If lastRow <= hdrRow Then Exit Sub

    For r = hdrRow + 1 To lastRow
        idv = Trim$(CStr(ws.Cells(r, colId).Value))
        If idv <> "" Then

            uuidTxt = V05_GetCellText(ws, r, colUUID)
            cuentaTxt = V05_GetCellText(ws, r, colCuenta)
            transTxt = V05_GetCellText(ws, r, colTrans)
            montoLinea = V05_GetVentaLineAmount(ws, r, colTotal, colImporte, colMonto, colPrecio, colCantidad)

            If transTxt <> "" Then
                groupKey = "T|" & V05_NormKey(transTxt)
                tipoGrupo = "NUMERO_TRANSACCION"
            Else
                groupKey = "I|" & V05_NormKey(idv)
                tipoGrupo = "ID_VENTA"
            End If

            If groupKey <> "" Then
                If Not dictTipo.Exists(groupKey) Then
                    dictTipo.Add groupKey, tipoGrupo
                    dictTrans.Add groupKey, transTxt
                    dictIds.Add groupKey, idv
                    dictUUIDs.Add groupKey, uuidTxt
                    dictMonto.Add groupKey, CDbl(0)
                    dictCuenta.Add groupKey, cuentaTxt
                Else
                    dictIds(groupKey) = V05_AppendUniqueToken(CStr(dictIds(groupKey)), idv)
                    dictUUIDs(groupKey) = V05_AppendUniqueToken(CStr(dictUUIDs(groupKey)), uuidTxt)

                    If Trim$(CStr(dictCuenta(groupKey))) = "" And cuentaTxt <> "" Then
                        dictCuenta(groupKey) = cuentaTxt
                    End If
                End If

                dictMonto(groupKey) = CDbl(dictMonto(groupKey)) + montoLinea
            End If
        End If
    Next r
End Sub


Private Sub V05_ContarAbonosBanco( _
    ByVal ws As Worksheet, _
    ByRef total As Long, _
    ByRef conId As Long, _

    Dim colCargo As Long, colAbono As Long, colId As Long
    Dim lastRow As Long, r As Long
    Dim cargoVal As Double, abonoVal As Double
    Dim idTxt As String

    Set hdrDia = ws.Cells.Find(What:="Día", LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)

Private Function V05_GetVentaLineAmount( _
    ByVal ws As Worksheet, _
    ByVal rowNum As Long, _
    ByVal colVTotal As Long, _
    ByVal colVImporte As Long, _
    ByVal colVMonto As Long, _
    ByVal colVPrecio As Long, _
    ByVal colVCantidad As Long) As Double

    Dim totalVal As Double
    Dim importeVal As Double
    Dim montoVal As Double
    Dim precioVal As Double
    Dim cantidadVal As Double

    If colVTotal > 0 Then
        totalVal = V05_ParseNumber(ws.Cells(rowNum, colVTotal).Value)
        If totalVal <> 0 Then
            V05_GetVentaLineAmount = totalVal
            Exit Function
        End If
    End If

    If colVImporte > 0 Then
        importeVal = V05_ParseNumber(ws.Cells(rowNum, colVImporte).Value)
        If importeVal <> 0 Then
            V05_GetVentaLineAmount = importeVal
            Exit Function
        End If
    End If

    If colVMonto > 0 Then
        montoVal = V05_ParseNumber(ws.Cells(rowNum, colVMonto).Value)
        If montoVal <> 0 Then
            V05_GetVentaLineAmount = montoVal
            Exit Function
        End If
    End If

    If colVPrecio > 0 Then
        precioVal = V05_ParseNumber(ws.Cells(rowNum, colVPrecio).Value)

        If colVCantidad > 0 Then
            cantidadVal = V05_ParseNumber(ws.Cells(rowNum, colVCantidad).Value)
            If cantidadVal = 0 Then cantidadVal = 1
            V05_GetVentaLineAmount = precioVal * cantidadVal
        Else
            V05_GetVentaLineAmount = precioVal
        End If
    End If
End Function


Private Function V05_AppendUniqueToken(ByVal existingList As String, ByVal token As String) As String
    Dim arr As Variant
    Dim i As Long
    Dim cleanToken As String

    cleanToken = Trim$(token)
    If cleanToken = "" Then
        V05_AppendUniqueToken = existingList
        Exit Function
    End If

    If Trim$(existingList) = "" Then
        V05_AppendUniqueToken = cleanToken
        Exit Function
    End If

    arr = Split(existingList, " | ")
    For i = LBound(arr) To UBound(arr)
        If StrComp(Trim$(CStr(arr(i))), cleanToken, vbTextCompare) = 0 Then
            V05_AppendUniqueToken = existingList
            Exit Function
        End If
    Next i

    V05_AppendUniqueToken = existingList & " | " & cleanToken
End Function


Private Function V05_GetCellText(ByVal ws As Worksheet, ByVal rowNum As Long, ByVal colNum As Long) As String
    If ws Is Nothing Then Exit Function
    If colNum <= 0 Then Exit Function
    V05_GetCellText = Trim$(CStr(ws.Cells(rowNum, colNum).Value))
End Function

Private Function V05_FindHeader(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headers As Variant) As Range
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim txt As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        txt = V05_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headers) To UBound(headers)
            If txt = V05_NormHeader(CStr(headers(i))) Then
                Set V05_FindHeader = ws.Cells(hdrRow, c)
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function V05_FindHeaderCol(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headerNames As Variant) As Long
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim cellText As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        cellText = V05_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headerNames) To UBound(headerNames)
            If cellText = V05_NormHeader(CStr(headerNames(i))) Then
                V05_FindHeaderCol = c
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function V05_NormHeader(ByVal s As String) As String
    s = UCase$(Trim$(s))
    s = Replace(s, "Á", "A")
    s = Replace(s, "É", "E")
    s = Replace(s, "Í", "I")
    s = Replace(s, "Ó", "O")
    s = Replace(s, "Ú", "U")
    s = Replace(s, "Ü", "U")
    s = Replace(s, ".", "")
    s = Replace(s, ",", "")
    s = Replace(s, "/", "")
    Do While InStr(s, "  ") > 0
        s = Replace(s, "  ", " ")
    Loop
    V05_NormHeader = s
End Function


Private Function V05_LastUsedRow(ByVal ws As Worksheet) As Long

    Set c = ws.Cells.Find(What:="*", _
                          After:=ws.Range("A1"), _
                          LookIn:=xlFormulas, _
                          LookAt:=xlPart, _
                          SearchOrder:=xlByRows, _
                          SearchDirection:=xlPrevious, _
                          MatchCase:=False)

    If c Is Nothing Then
        V05_LastUsedRow = 0
    Else
        V05_LastUsedRow = c.Row
    End If
End Function


Private Function V05_ParseNumber(ByVal v As Variant) As Double
    Dim s As String
    Dim neg As Boolean

    On Error GoTo ErrHandler

    If IsNumeric(v) Then
        V05_ParseNumber = CDbl(v)
        Exit Function
    End If

    s = Trim$(CStr(v))
    If s = "" Then Exit Function

    s = Replace(s, Chr(160), "")
    s = Replace(s, "$", "")
    s = Replace(s, ",", "")
    s = Replace(s, " ", "")

    If Left$(s, 1) = "(" And Right$(s, 1) = ")" Then
        neg = True
        s = Replace(s, "(", "")
        s = Replace(s, ")", "")
    End If

    If IsNumeric(s) Then
        V05_ParseNumber = CDbl(s)
        If neg Then V05_ParseNumber = -V05_ParseNumber
    End If
    Exit Function

ErrHandler:
    V05_ParseNumber = 0
End Function

Private Sub V06_ContarVentasAgrupadas( _
    ByVal ws As Worksheet, _
    ByRef total As Long, _
    ByRef conCuenta As Long, _
    ByRef sinCuenta As Long)

    Dim dictTipo As Object, dictTrans As Object, dictIds As Object, dictUUIDs As Object, dictMonto As Object, dictCuenta As Object

    Set dictTipo = CreateObject("Scripting.Dictionary")
    Set dictTrans = CreateObject("Scripting.Dictionary")
    Set dictIds = CreateObject("Scripting.Dictionary")
    Set dictUUIDs = CreateObject("Scripting.Dictionary")
    Set dictMonto = CreateObject("Scripting.Dictionary")
    Set dictCuenta = CreateObject("Scripting.Dictionary")

    V06_BuildVentasGroupingData ws, dictTipo, dictTrans, dictIds, dictUUIDs, dictMonto, dictCuenta

    Dim k As Variant
    For Each k In dictTipo.Keys
        total = total + 1

        Else

        End If

Private Sub V06_BuildVentasGroupingData( _
    ByVal ws As Worksheet, _
    ByVal dictTipo As Object, _
    ByVal dictTrans As Object, _
    ByVal dictIds As Object, _
    ByVal dictUUIDs As Object, _
    ByVal dictMonto As Object, _
    ByVal dictCuenta As Object)

    Dim hdrRow As Long
    Dim colId As Long, colUUID As Long, colCuenta As Long, colTrans As Long
    Dim colTotal As Long, colImporte As Long, colMonto As Long, colPrecio As Long, colCantidad As Long
    Dim lastRow As Long, r As Long

    Dim idv As String, uuidTxt As String, cuentaTxt As String, transTxt As String
    Dim groupKey As String, tipoGrupo As String
    Dim montoLinea As Double

    If ws Is Nothing Then Exit Sub

    hdrRow = V06_FindHeaderRowVentas(ws)
    If hdrRow = 0 Then Exit Sub

    colId = V06_FindHeaderCol(ws, hdrRow, Array("ID_venta", "id_venta"))
    colUUID = V06_FindHeaderCol(ws, hdrRow, Array("UUID", "uuid"))
    colCuenta = V06_FindHeaderCol(ws, hdrRow, Array("Cuenta bancaria"))
    colTrans = V06_FindHeaderCol(ws, hdrRow, Array("numero_transaccion", "Número de transacción", "Numero de transaccion", "numero transaccion", "Numero transaccion", "Operacion relacionada"))

    colTotal = V06_FindHeaderCol(ws, hdrRow, Array("Total", "total"))
    colImporte = V06_FindHeaderCol(ws, hdrRow, Array("Importe", "importe"))
    colMonto = V06_FindHeaderCol(ws, hdrRow, Array("Monto", "monto", "Monto total", "monto total", "Total venta", "total venta", "Valor de la operación", "Valor de la operacion"))
    colPrecio = V06_FindHeaderCol(ws, hdrRow, Array("precio_unitario", "precio unitario", "Precio_Unitario", "Precio Unitario"))
    colCantidad = V06_FindHeaderCol(ws, hdrRow, Array("Cantidad", "cantidad", "Qty", "qty", "Cant", "cant"))

    If colId = 0 Then Exit Sub

    lastRow = V06_LastUsedRow(ws)
    If lastRow <= hdrRow Then Exit Sub

    For r = hdrRow + 1 To lastRow
        idv = Trim$(CStr(ws.Cells(r, colId).Value))
        If idv <> "" Then

            uuidTxt = V06_GetCellText(ws, r, colUUID)
            cuentaTxt = V06_GetCellText(ws, r, colCuenta)
            transTxt = V06_GetCellText(ws, r, colTrans)
            montoLinea = V06_GetVentaLineAmount(ws, r, colTotal, colImporte, colMonto, colPrecio, colCantidad)

            If transTxt <> "" Then
                groupKey = "T|" & V06_NormKey(transTxt)
                tipoGrupo = "NUMERO_TRANSACCION"
            Else
                groupKey = "I|" & V06_NormKey(idv)
                tipoGrupo = "ID_VENTA"
            End If

            If groupKey <> "" Then
                If Not dictTipo.Exists(groupKey) Then
                    dictTipo.Add groupKey, tipoGrupo
                    dictTrans.Add groupKey, transTxt
                    dictIds.Add groupKey, idv
                    dictUUIDs.Add groupKey, uuidTxt
                    dictMonto.Add groupKey, CDbl(0)
                    dictCuenta.Add groupKey, cuentaTxt
                Else
                    dictIds(groupKey) = V06_AppendUniqueToken(CStr(dictIds(groupKey)), idv)
                    dictUUIDs(groupKey) = V06_AppendUniqueToken(CStr(dictUUIDs(groupKey)), uuidTxt)

                    If Trim$(CStr(dictCuenta(groupKey))) = "" And cuentaTxt <> "" Then
                        dictCuenta(groupKey) = cuentaTxt
                    End If
                End If

                dictMonto(groupKey) = CDbl(dictMonto(groupKey)) + montoLinea
            End If
        End If
    Next r
End Sub


Private Sub V06_ContarAbonosBanco( _
    ByVal ws As Worksheet, _
    ByRef total As Long, _
    ByRef conId As Long, _
    ByRef sinId As Long)

    Dim hdrDia As Range, hdrCargo As Range, hdrAbono As Range
    Dim hdrRow As Long
    Dim colCargo As Long, colAbono As Long, colId As Long
    Dim lastRow As Long, r As Long
    Dim cargoVal As Double, abonoVal As Double
    Dim idTxt As String

    Set hdrDia = ws.Cells.Find(What:="Día", LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    If hdrDia Is Nothing Then
        Set hdrDia = ws.Cells.Find(What:="Dia", LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    End If
    If hdrDia Is Nothing Then Exit Sub

    hdrRow = hdrDia.Row

    Set hdrCargo = V06_FindHeader(ws, hdrRow, Array("Cargo"))
    Set hdrAbono = V06_FindHeader(ws, hdrRow, Array("Abono"))

    If hdrCargo Is Nothing Or hdrAbono Is Nothing Then Exit Sub

    colCargo = hdrCargo.Column
    colAbono = hdrAbono.Column
    colId = V06_FindHeaderCol(ws, hdrRow, Array("ID_venta", "id_venta"))

    lastRow = V06_LastUsedRow(ws)
    If lastRow <= hdrRow Then Exit Sub

    For r = hdrRow + 1 To lastRow
        cargoVal = V06_ParseNumber(ws.Cells(r, colCargo).Value)
        abonoVal = V06_ParseNumber(ws.Cells(r, colAbono).Value)

        If abonoVal > 0 And cargoVal = 0 Then
            total = total + 1
            idTxt = V06_GetCellText(ws, r, colId)

            If idTxt <> "" Then
                conId = conId + 1
            Else
                sinId = sinId + 1
            End If
        End If
    Next r
End Sub

    Dim hdrRow As Long
    Dim colDia As Long, colCargo As Long, colAbono As Long, colConcepto As Long

    End If
    If hdrDia Is Nothing Then Exit Sub

    hdrRow = hdrDia.Row
    colDia = hdrDia.Column

    Set hdrCargo = V06_FindHeader(ws, hdrRow, Array("Cargo"))

    colObs = V06_FindHeaderCol(ws, hdrRow, Array("OBS"))

Private Function V06_GetVentaLineAmount( _
    ByVal ws As Worksheet, _
    ByVal rowNum As Long, _
    ByVal colVTotal As Long, _
    ByVal colVImporte As Long, _
    ByVal colVMonto As Long, _
    ByVal colVPrecio As Long, _
    ByVal colVCantidad As Long) As Double

    Dim totalVal As Double
    Dim importeVal As Double
    Dim montoVal As Double
    Dim precioVal As Double
    Dim cantidadVal As Double

    If colVTotal > 0 Then
        totalVal = V06_ParseNumber(ws.Cells(rowNum, colVTotal).Value)
        If totalVal <> 0 Then
            V06_GetVentaLineAmount = totalVal
            Exit Function
        End If
    End If

    If colVImporte > 0 Then
        importeVal = V06_ParseNumber(ws.Cells(rowNum, colVImporte).Value)
        If importeVal <> 0 Then
            V06_GetVentaLineAmount = importeVal
            Exit Function
        End If
    End If

    If colVMonto > 0 Then
        montoVal = V06_ParseNumber(ws.Cells(rowNum, colVMonto).Value)
        If montoVal <> 0 Then
            V06_GetVentaLineAmount = montoVal
            Exit Function
        End If
    End If

    If colVPrecio > 0 Then
        precioVal = V06_ParseNumber(ws.Cells(rowNum, colVPrecio).Value)

        If colVCantidad > 0 Then
            cantidadVal = V06_ParseNumber(ws.Cells(rowNum, colVCantidad).Value)
            If cantidadVal = 0 Then cantidadVal = 1
            V06_GetVentaLineAmount = precioVal * cantidadVal
        Else
            V06_GetVentaLineAmount = precioVal
        End If
    End If
End Function


Private Function V06_AppendUniqueToken(ByVal existingList As String, ByVal token As String) As String
    Dim arr As Variant
    Dim i As Long
    Dim cleanToken As String

    cleanToken = Trim$(token)
    If cleanToken = "" Then
        V06_AppendUniqueToken = existingList
        Exit Function
    End If

    If Trim$(existingList) = "" Then
        V06_AppendUniqueToken = cleanToken
        Exit Function
    End If

    arr = Split(existingList, " | ")
    For i = LBound(arr) To UBound(arr)
        If StrComp(Trim$(CStr(arr(i))), cleanToken, vbTextCompare) = 0 Then
            V06_AppendUniqueToken = existingList
            Exit Function
        End If
    Next i

    V06_AppendUniqueToken = existingList & " | " & cleanToken
End Function

Private Function V06_FindHeader(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headers As Variant) As Range
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim txt As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        txt = V06_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headers) To UBound(headers)
            If txt = V06_NormHeader(CStr(headers(i))) Then
                Set V06_FindHeader = ws.Cells(hdrRow, c)
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function V06_FindHeaderCol(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headerNames As Variant) As Long
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim cellText As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        cellText = V06_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headerNames) To UBound(headerNames)
            If cellText = V06_NormHeader(CStr(headerNames(i))) Then
                V06_FindHeaderCol = c
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function V06_NormHeader(ByVal s As String) As String
    s = UCase$(Trim$(s))
    s = Replace(s, "Á", "A")
    s = Replace(s, "É", "E")
    s = Replace(s, "Í", "I")
    s = Replace(s, "Ó", "O")
    s = Replace(s, "Ú", "U")
    s = Replace(s, "Ü", "U")
    s = Replace(s, ".", "")
    s = Replace(s, ",", "")
    s = Replace(s, "/", "")
    Do While InStr(s, "  ") > 0
        s = Replace(s, "  ", " ")
    Loop
    V06_NormHeader = s
End Function


Private Function V06_LastUsedRow(ByVal ws As Worksheet) As Long
    Dim c As Range

    Set c = ws.Cells.Find(What:="*", _
                          After:=ws.Range("A1"), _
                          LookIn:=xlFormulas, _
                          LookAt:=xlPart, _
                          SearchOrder:=xlByRows, _
                          SearchDirection:=xlPrevious, _
                          MatchCase:=False)

    If c Is Nothing Then
        V06_LastUsedRow = 0
    Else
        V06_LastUsedRow = c.Row
    End If
End Function


Private Function V06_ParseNumber(ByVal v As Variant) As Double
    Dim s As String
    Dim neg As Boolean

    On Error GoTo ErrHandler

    If IsNumeric(v) Then
        V06_ParseNumber = CDbl(v)
        Exit Function
    End If

    s = Trim$(CStr(v))
    If s = "" Then Exit Function

    s = Replace(s, Chr(160), "")
    s = Replace(s, "$", "")
    s = Replace(s, ",", "")
    s = Replace(s, " ", "")

    If Left$(s, 1) = "(" And Right$(s, 1) = ")" Then
        neg = True
        s = Replace(s, "(", "")
        s = Replace(s, ")", "")
    End If

    If IsNumeric(s) Then
        V06_ParseNumber = CDbl(s)
        If neg Then V06_ParseNumber = -V06_ParseNumber
    End If
    Exit Function

ErrHandler:
    V06_ParseNumber = 0
End Function

Private Function V07_SumarBanco(ByVal ws As Worksheet, ByRef totalCargo As Double, ByRef totalAbono As Double) As Boolean

    Dim hdrDia As Range, hdrCargo As Range, hdrAbono As Range
    Dim hdrRow As Long
    Dim colCargo As Long, colAbono As Long
    Dim lastRow As Long, r As Long

    Set hdrDia = ws.Cells.Find(What:="Día", LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    If hdrDia Is Nothing Then
        Set hdrDia = ws.Cells.Find(What:="Dia", LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    End If
    If hdrDia Is Nothing Then Exit Function

    hdrRow = hdrDia.Row

    Set hdrCargo = V07_FindHeader(ws, hdrRow, Array("Cargo"))
    Set hdrAbono = V07_FindHeader(ws, hdrRow, Array("Abono"))

    If hdrCargo Is Nothing Or hdrAbono Is Nothing Then Exit Function

    colCargo = hdrCargo.Column
    colAbono = hdrAbono.Column

    lastRow = V07_LastUsedRow(ws)
    If lastRow <= hdrRow Then
        V07_SumarBanco = True
        Exit Function
    End If

    For r = hdrRow + 1 To lastRow
        totalCargo = totalCargo + V07_ParseNumber(ws.Cells(r, colCargo).Value)
        totalAbono = totalAbono + V07_ParseNumber(ws.Cells(r, colAbono).Value)
    Next r

    V07_SumarBanco = True
End Function


Private Function V07_SumarEstMP( _
    ByVal ws As Worksheet, _
    ByRef totalCargo As Double, _
    ByRef totalAbono As Double, _
    ByRef totalValor As Double) As Boolean

    Dim hdrRow As Long
    Dim colValor As Long, colCargo As Long, colAbono As Long, colTipo As Long
    Dim lastRow As Long, r As Long
    Dim valOperacion As Double
    Dim txtTipo As String

    hdrRow = V07_FindHeaderRowEstMP(ws)
    If hdrRow = 0 Then Exit Function

    colValor = V07_FindHeaderCol(ws, hdrRow, Array("Valor de la operación", "Valor de la operacion", "Valor operación", "Valor operacion"))
    colCargo = V07_FindHeaderCol(ws, hdrRow, Array("Cargo"))
    colAbono = V07_FindHeaderCol(ws, hdrRow, Array("Abono"))
    colTipo = V07_FindHeaderCol(ws, hdrRow, Array("Tipo de operación", "Tipo de operacion", "Tipo operación", "Tipo operacion", "Tipo"))

    If colValor = 0 And colCargo = 0 And colAbono = 0 Then Exit Function

    lastRow = V07_LastUsedRow(ws)
    If lastRow <= hdrRow Then
        V07_SumarEstMP = True
        Exit Function
    End If

    For r = hdrRow + 1 To lastRow

        If colCargo > 0 Then
            totalCargo = totalCargo + V07_ParseNumber(ws.Cells(r, colCargo).Value)
        End If

        If colAbono > 0 Then
            totalAbono = totalAbono + V07_ParseNumber(ws.Cells(r, colAbono).Value)
        End If

        If colValor > 0 Then
            valOperacion = V07_ParseNumber(ws.Cells(r, colValor).Value)
            totalValor = totalValor + valOperacion

            If colCargo = 0 And colAbono = 0 Then

                txtTipo = ""
                If colTipo > 0 Then txtTipo = V07_NormHeader(CStr(ws.Cells(r, colTipo).Value))

                If txtTipo <> "" Then
                    If InStr(1, txtTipo, "CARGO", vbTextCompare) > 0 _
                       Or InStr(1, txtTipo, "DEBIT", vbTextCompare) > 0 _
                       Or InStr(1, txtTipo, "RETIRO", vbTextCompare) > 0 _
                       Or InStr(1, txtTipo, "PAGO", vbTextCompare) > 0 Then

                        If valOperacion <> 0 Then totalCargo = totalCargo + Abs(valOperacion)

                    ElseIf InStr(1, txtTipo, "ABONO", vbTextCompare) > 0 _
                       Or InStr(1, txtTipo, "CREDIT", vbTextCompare) > 0 _
                       Or InStr(1, txtTipo, "INGRESO", vbTextCompare) > 0 _
                       Or InStr(1, txtTipo, "COBRO", vbTextCompare) > 0 _
                       Or InStr(1, txtTipo, "LIQUID", vbTextCompare) > 0 Then

                        If valOperacion <> 0 Then totalAbono = totalAbono + Abs(valOperacion)

                    Else
                        If valOperacion < 0 Then
                            totalCargo = totalCargo + Abs(valOperacion)
                        ElseIf valOperacion > 0 Then
                            totalAbono = totalAbono + valOperacion
                        End If
                    End If
                Else
                    If valOperacion < 0 Then
                         totalCargo = totalCargo + Abs(valOperacion)
                    ElseIf valOperacion > 0 Then
                        totalAbono = totalAbono + valOperacion
                    End If
                End If
            End If
        End If
    Next r

    V07_SumarEstMP = True
End Function

    Dim r As Long

Private Function V07_FindHeader(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headers As Variant) As Range
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim txt As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        txt = V07_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headers) To UBound(headers)
            If txt = V07_NormHeader(CStr(headers(i))) Then
                 Set V07_FindHeader = ws.Cells(hdrRow, c)
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function V07_FindHeaderCol(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headerNames As Variant) As Long
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim cellText As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        cellText = V07_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headerNames) To UBound(headerNames)
            If cellText = V07_NormHeader(CStr(headerNames(i))) Then
                V07_FindHeaderCol = c
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function V07_NormHeader(ByVal s As String) As String
    s = UCase$(Trim$(s))
    s = Replace(s, "Á", "A")
    s = Replace(s, "É", "E")
    s = Replace(s, "Í", "I")
    s = Replace(s, "Ó", "O")
    s = Replace(s, "Ú", "U")
    s = Replace(s, "Ü", "U")
    s = Replace(s, ".", "")
    s = Replace(s, ",", "")
    s = Replace(s, "/", "")
    Do While InStr(s, "  ") > 0
        s = Replace(s, "  ", " ")
    Loop
    V07_NormHeader = s
End Function


Private Function V07_LastUsedRow(ByVal ws As Worksheet) As Long
    Dim c As Range

    Set c = ws.Cells.Find(What:="*", _
                          After:=ws.Range("A1"), _
                           LookIn:=xlFormulas, _
                          LookAt:=xlPart, _
                          SearchOrder:=xlByRows, _
                          SearchDirection:=xlPrevious, _
                           MatchCase:=False)

    If c Is Nothing Then
        V07_LastUsedRow = 0
    Else
        V07_LastUsedRow = c.Row
    End If
End Function


Private Function V07_ParseNumber(ByVal v As Variant) As Double
    Dim s As String
    Dim neg As Boolean

    On Error GoTo ErrHandler

    If IsNumeric(v) Then
         V07_ParseNumber = CDbl(v)
        Exit Function
    End If

    s = Trim$(CStr(v))
    If s = "" Then Exit Function

    s = Replace(s, Chr(160), "")
    s = Replace(s, "$", "")
    s = Replace(s, ",", "")
    s = Replace(s, " ", "")

    If Left$(s, 1) = "(" And Right$(s, 1) = ")" Then
        neg = True
        s = Replace(s, "(", "")
        s = Replace(s, ")", "")
    End If

    If IsNumeric(s) Then
        V07_ParseNumber = CDbl(s)
        If neg Then V07_ParseNumber = -V07_ParseNumber
    End If
    Exit Function

ErrHandler:
    V07_ParseNumber = 0
End Function

Private Function V08_ContarBanco( _
    ByVal ws As Worksheet, _
    ByRef movs As Long, _
    ByRef totalCargo As Double, _
    ByRef totalAbono As Double, _
    ByRef conObs As Long, _
    ByRef sinObs As Long, _
    ByRef conPUE As Long, _
    ByRef conPPD As Long, _
    ByRef conId As Long, _
    ByRef observacion As String) As Boolean

    Dim hdrDia As Range, hdrCargo As Range, hdrAbono As Range
    Dim hdrRow As Long
    Dim colCargo As Long, colAbono As Long, colObs As Long, colCompl As Long, colId As Long
    Dim lastRow As Long, r As Long
    Dim cargoVal As Double, abonoVal As Double
    Dim obsTxt As String, complTxt As String, idTxt As String

    Set hdrDia = ws.Cells.Find(What:="Día", LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    If hdrDia Is Nothing Then
        Set hdrDia = ws.Cells.Find(What:="Dia", LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    End If
    If hdrDia Is Nothing Then
        observacion = "NO SE ENCONTRÓ COLUMNA DÍA"
        Exit Function
    End If

    hdrRow = hdrDia.Row

    Set hdrCargo = V08_FindHeader(ws, hdrRow, Array("Cargo"))
    Set hdrAbono = V08_FindHeader(ws, hdrRow, Array("Abono"))

    If hdrCargo Is Nothing Or hdrAbono Is Nothing Then
        observacion = "NO SE ENCONTRARON CARGO / ABONO"
        Exit Function
    End If

    colCargo = hdrCargo.Column
    colAbono = hdrAbono.Column
    colObs = V08_FindHeaderCol(ws, hdrRow, Array("OBS"))
    colCompl = V08_FindHeaderCol(ws, hdrRow, Array("COMPL. CTA", "COMPL CTA", "COMPL_CTA"))
    colId = V08_FindHeaderCol(ws, hdrRow, Array("ID_venta", "id_venta"))

    lastRow = V08_LastUsedRow(ws)
    If lastRow <= hdrRow Then
        V08_ContarBanco = True
        Exit Function
    End If

    For r = hdrRow + 1 To lastRow
        cargoVal = V08_ParseNumber(ws.Cells(r, colCargo).Value)
        abonoVal = V08_ParseNumber(ws.Cells(r, colAbono).Value)

        If cargoVal <> 0 Or abonoVal <> 0 Then
            movs = movs + 1
            totalCargo = totalCargo + cargoVal
            totalAbono = totalAbono + abonoVal

            obsTxt = V08_GetCellText(ws, r, colObs)
            complTxt = V08_NormHeader(V08_GetCellText(ws, r, colCompl))
            idTxt = V08_GetCellText(ws, r, colId)

            If obsTxt <> "" Then
                conObs = conObs + 1
            Else
                sinObs = sinObs + 1
            End If

            If InStr(1, complTxt, "PUE", vbTextCompare) > 0 Then conPUE = conPUE + 1
            If InStr(1, complTxt, "PPD", vbTextCompare) > 0 Then conPPD = conPPD + 1
            If idTxt <> "" Then conId = conId + 1
        End If
    Next r

    V08_ContarBanco = True
End Function

Private Function V08_ContarVentasMP( _
    ByVal ws As Worksheet, _
    ByRef totalReg As Long, _
    ByRef conMatch As Long, _
    ByRef sinMatch As Long, _
    ByRef conUUID As Long, _
    ByRef sinUUID As Long, _
    ByRef conTrans As Long, _
    ByRef sinTrans As Long, _
    ByRef observacion As String) As Boolean

    Dim hdrRow As Long, lastRow As Long, r As Long
    Dim colOK As Long, colUUID As Long, colTrans As Long, colId As Long
    Dim okTxt As String, uuidTxt As String, transTxt As String
    Dim filaTieneDato As Boolean

    If ws Is Nothing Then
        observacion = "NO EXISTE LA HOJA"
        Exit Function
    End If

    hdrRow = V08_FindHeaderRowVentasMP(ws)
    If hdrRow = 0 Then
        observacion = "NO SE IDENTIFICARON ENCABEZADOS"
        Exit Function
    End If

    colOK = V08_FindHeaderCol(ws, hdrRow, Array("OK_MP", "OK MP"))
    colUUID = V08_FindHeaderCol(ws, hdrRow, Array("UUID", "uuid"))
    colTrans = V08_FindHeaderCol(ws, hdrRow, Array("numero_transaccion", "Número de transacción", "Numero de transaccion", "Operacion relacionada"))
    colId = V08_FindHeaderCol(ws, hdrRow, Array("ID_venta", "id_venta"))

    lastRow = V08_LastUsedRow(ws)
    If lastRow <= hdrRow Then
        V08_ContarVentasMP = True
        Exit Function
    End If

    For r = hdrRow + 1 To lastRow
        filaTieneDato = V08_RowHasData(ws, r, Array(colId, colUUID, colTrans, colOK))
        If filaTieneDato Then
            totalReg = totalReg + 1

            okTxt = V08_GetCellText(ws, r, colOK)
            uuidTxt = V08_GetCellText(ws, r, colUUID)
            transTxt = V08_GetCellText(ws, r, colTrans)

            If V08_IsOKMatch(okTxt) Then
                conMatch = conMatch + 1
            Else
                sinMatch = sinMatch + 1
            End If

            If uuidTxt <> "" Then
                conUUID = conUUID + 1
            Else
                sinUUID = sinUUID + 1
            End If

            If transTxt <> "" Then
                conTrans = conTrans + 1
            Else
                sinTrans = sinTrans + 1
            End If
        End If
    Next r

    V08_ContarVentasMP = True
End Function

        V08_IsOKMatch = True
    End If
End Function

Private Function V08_ContarVentasAgrupadas( _
    ByVal sheetName As String, _
    ByVal usarTrans As Boolean, _
    ByRef totalGrupos As Long, _
    ByRef conCuenta As Long, _
    ByRef sinCuenta As Long, _
    ByRef conUUID As Long, _
    ByRef sinUUID As Long, _
    ByRef observacion As String) As Boolean

    Dim ws As Worksheet
    Dim hdrRow As Long, lastRow As Long, r As Long
    Dim colId As Long, colUUID As Long, colCuenta As Long, colTrans As Long
    Dim dictCuenta As Object, dictUUID As Object
    Dim groupKey As String, idTxt As String, uuidTxt As String, cuentaTxt As String, transTxt As String
    Dim k As Variant

    Set ws = V08_GetSheet(sheetName)
    If ws Is Nothing Then
        observacion = "NO EXISTE LA HOJA"
        Exit Function
    End If

    hdrRow = V08_FindHeaderRowVentasAgrupadas(ws)
    If hdrRow = 0 Then
        observacion = "NO SE IDENTIFICARON ENCABEZADOS"
        Exit Function
    End If

    colId = V08_FindHeaderCol(ws, hdrRow, Array("ID_venta", "id_venta"))
    colUUID = V08_FindHeaderCol(ws, hdrRow, Array("UUID", "uuid"))
    colCuenta = V08_FindHeaderCol(ws, hdrRow, Array("Cuenta bancaria"))
    colTrans = V08_FindHeaderCol(ws, hdrRow, Array("numero_transaccion", "Número de transacción", "Numero de transaccion", "numero transaccion", "Numero transaccion", "Operacion relacionada"))

    If colId = 0 Then
        observacion = "NO EXISTE ID_VENTA"
        Exit Function
    End If

    Set dictCuenta = CreateObject("Scripting.Dictionary")
    Set dictUUID = CreateObject("Scripting.Dictionary")

    lastRow = V08_LastUsedRow(ws)
    If lastRow <= hdrRow Then
        V08_ContarVentasAgrupadas = True
        Exit Function
    End If

    For r = hdrRow + 1 To lastRow
        idTxt = V08_GetCellText(ws, r, colId)
        If idTxt <> "" Then

            uuidTxt = V08_GetCellText(ws, r, colUUID)
            cuentaTxt = V08_GetCellText(ws, r, colCuenta)
            transTxt = V08_GetCellText(ws, r, colTrans)

            If usarTrans And transTxt <> "" Then
                groupKey = "T|" & V08_NormKey(transTxt)
            Else
                groupKey = "I|" & V08_NormKey(idTxt)
            End If

            If groupKey <> "" Then
                If Not dictCuenta.Exists(groupKey) Then
                    dictCuenta.Add groupKey, cuentaTxt
                    dictUUID.Add groupKey, uuidTxt
                Else
                    If Trim$(CStr(dictCuenta(groupKey))) = "" And cuentaTxt <> "" Then
                        dictCuenta(groupKey) = cuentaTxt
                    End If
                    dictUUID(groupKey) = V08_AppendUniqueToken(CStr(dictUUID(groupKey)), uuidTxt)
                End If
            End If
        End If
    Next r

    For Each k In dictCuenta.Keys
        totalGrupos = totalGrupos + 1

        If Trim$(CStr(dictCuenta(k))) <> "" Then
            conCuenta = conCuenta + 1
        Else
            sinCuenta = sinCuenta + 1
        End If

        If Trim$(CStr(dictUUID(k))) <> "" Then
            conUUID = conUUID + 1
        Else
            sinUUID = sinUUID + 1
        End If
    Next k

    V08_ContarVentasAgrupadas = True
End Function

    For r = 1 To 10
        If V08_FindHeaderCol(ws, r, Array("ID_venta", "id_venta")) > 0 Then
            V08_FindHeaderRowVentasAgrupadas = r
            Exit Function
        End If
    Next r
End Function

        wsOut.Range("A" & outRow + 4 & ":G" & outRow + 4).Value = Array(IIf(V08_FindPagosESheetName = "", "PAGOS E / COMPL. E", V08_FindPagosESheetName), totalBase, conCuenta, sinCuenta, conId, sinId, "OK")
    Else
        wsOut.Range("A" & outRow + 4 & ":G" & outRow + 4).Value = Array("PAGOS E / COMPL. E", 0, 0, 0, 0, 0, obs)
    End If

    If V08_ContarDocumentoPorCobertura("CFDI_E_PPD", totalBase, conCuenta, sinCuenta, conId, sinId, obs) Then
        wsOut.Range("A" & outRow + 5 & ":G" & outRow + 5).Value = Array("CFDI_E_PPD", totalBase, conCuenta, sinCuenta, conId, sinId, "OK")
    Else
        wsOut.Range("A" & outRow + 5 & ":G" & outRow + 5).Value = Array("CFDI_E_PPD", 0, 0, 0, 0, 0, obs)
    End If

    If V08_ContarDocumentoPorCobertura("CFDI_I_PUE", totalBase, conCuenta, sinCuenta, conId, sinId, obs) Then
        wsOut.Range("A" & outRow + 6 & ":G" & outRow + 6).Value = Array("CFDI_I_PUE", totalBase, conCuenta, sinCuenta, conId, sinId, "OK")
    Else
        wsOut.Range("A" & outRow + 6 & ":G" & outRow + 6).Value = Array("CFDI_I_PUE", 0, 0, 0, 0, 0, obs)
    End If

    If V08_ContarDocumentoPorCobertura("CFDI_I_PPD", totalBase, conCuenta, sinCuenta, conId, sinId, obs) Then
        wsOut.Range("A" & outRow + 7 & ":G" & outRow + 7).Value = Array("CFDI_I_PPD", totalBase, conCuenta, sinCuenta, conId, sinId, "OK")
    Else
        wsOut.Range("A" & outRow + 7 & ":G" & outRow + 7).Value = Array("CFDI_I_PPD", 0, 0, 0, 0, 0, obs)
    End If

    If V08_ContarDocumentoPorCobertura(V08_FindPagosISheetName, totalBase, conCuenta, sinCuenta, conId, sinId, obs) Then
        wsOut.Range("A" & outRow + 8 & ":G" & outRow + 8).Value = Array(IIf(V08_FindPagosISheetName = "", "PAGOS I / COMPL. I", V08_FindPagosISheetName), totalBase, conCuenta, sinCuenta, conId, sinId, "OK")
    Else
        wsOut.Range("A" & outRow + 8 & ":G" & outRow + 8).Value = Array("PAGOS I / COMPL. I", 0, 0, 0, 0, 0, obs)
    End If

    outRow = outRow + 8
End Sub


Private Function V08_ContarDocumentoPorCobertura( _
    ByVal sheetName As String, _
    ByRef totalBase As Long, _
    ByRef conCuenta As Long, _
    ByRef sinCuenta As Long, _
    ByRef conId As Long, _
    ByRef sinId As Long, _
    ByRef observacion As String) As Boolean

    Dim ws As Worksheet
    Dim hdrRow As Long
    Dim colUUID As Long, colMother As Long, colCuenta As Long, colId As Long
    Dim lastRow As Long, r As Long
    Dim baseTxt As String, cuentaTxt As String, idTxt As String

    If Trim$(sheetName) = "" Then
        observacion = "NO EXISTE LA HOJA"
        Exit Function
    End If

    Set ws = V08_GetSheet(sheetName)
    If ws Is Nothing Then
        observacion = "NO EXISTE LA HOJA"
        Exit Function
    End If

    hdrRow = V08_FindHeaderRowDoc(ws)
    If hdrRow = 0 Then
        observacion = "NO SE IDENTIFICARON ENCABEZADOS"
        Exit Function
    End If

    colUUID = V08_FindHeaderCol(ws, hdrRow, Array("UUID", "uuid"))
    colMother = V08_FindHeaderCol(ws, hdrRow, Array("UUID Madre", "UUID MADRE", "UUID_MADRE"))
    colCuenta = V08_FindHeaderCol(ws, hdrRow, Array("Cuenta bancaria"))
    colId = V08_FindHeaderCol(ws, hdrRow, Array("ID_venta", "id_venta"))

    If colUUID = 0 And colMother = 0 Then
        observacion = "NO EXISTE UUID / UUID MADRE"
        Exit Function
    End If

    lastRow = V08_LastUsedRow(ws)
    If lastRow <= hdrRow Then
        V08_ContarDocumentoPorCobertura = True
        Exit Function
    End If

    For r = hdrRow + 1 To lastRow
        baseTxt = ""
        If colUUID > 0 Then baseTxt = V08_GetCellText(ws, r, colUUID)
        If baseTxt = "" And colMother > 0 Then baseTxt = V08_GetCellText(ws, r, colMother)

        If baseTxt <> "" Then
            totalBase = totalBase + 1

            cuentaTxt = V08_GetCellText(ws, r, colCuenta)
            idTxt = V08_GetCellText(ws, r, colId)

            If cuentaTxt <> "" Then
                conCuenta = conCuenta + 1
            Else
                sinCuenta = sinCuenta + 1
             End If

            If colId > 0 Then
                If idTxt <> "" Then
                    conId = conId + 1
                Else
                    sinId = sinId + 1
                End If
            End If
        End If
    Next r

    V08_ContarDocumentoPorCobertura = True
End Function


Private Function V08_FindHeaderRowDoc(ByVal ws As Worksheet) As Long
    Dim r As Long

    For r = 1 To 15
        If V08_FindHeaderCol(ws, r, Array("UUID", "uuid")) > 0 _
            Or V08_FindHeaderCol(ws, r, Array("UUID Madre", "UUID MADRE", "UUID_MADRE")) > 0 Then
            V08_FindHeaderRowDoc = r
            Exit Function
        End If
    Next r
End Function


Private Function V08_FindPagosESheetName() As String
    Dim arr, i As Long
    arr = Array("PAGOS E", "COMPL. E", "COMPL E", "PAGOS_E", "COMPL_E")

    For i = LBound(arr) To UBound(arr)
        If Not V08_GetSheet(CStr(arr(i))) Is Nothing Then
            V08_FindPagosESheetName = CStr(arr(i))
            Exit Function
        End If
    Next i
End Function


Private Function V08_FindPagosISheetName() As String
    Dim arr, i As Long
    arr = Array("PAGOS I", "COMPL. I", "COMPL I", "PAGOS_I", "COMPL_I")

    For i = LBound(arr) To UBound(arr)
        If Not V08_GetSheet(CStr(arr(i))) Is Nothing Then
            V08_FindPagosISheetName = CStr(arr(i))
            Exit Function
        End If
    Next i
End Function


Private Function V08_GetSheet(ByVal sheetName As String) As Worksheet
    On Error Resume Next
    Set V08_GetSheet = ThisWorkbook.Worksheets(sheetName)
    On Error GoTo 0
End Function


Private Function V08_FindHeader(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headers As Variant) As Range
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim txt As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        txt = V08_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headers) To UBound(headers)
            If txt = V08_NormHeader(CStr(headers(i))) Then
                Set V08_FindHeader = ws.Cells(hdrRow, c)
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function V08_FindHeaderCol(ByVal ws As Worksheet, ByVal hdrRow As Long, ByVal headerNames As Variant) As Long
    Dim c As Long, i As Long
    Dim lastCol As Long
    Dim cellText As String

    lastCol = ws.Cells(hdrRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        cellText = V08_NormHeader(CStr(ws.Cells(hdrRow, c).Value))
        For i = LBound(headerNames) To UBound(headerNames)
            If cellText = V08_NormHeader(CStr(headerNames(i))) Then
                V08_FindHeaderCol = c
                Exit Function
            End If
        Next i
    Next c
End Function


Private Function V08_NormHeader(ByVal s As String) As String
    s = UCase$(Trim$(s))
    s = Replace(s, "Á", "A")
    s = Replace(s, "É", "E")
    s = Replace(s, "Í", "I")
    s = Replace(s, "Ó", "O")
    s = Replace(s, "Ú", "U")
    s = Replace(s, "Ü", "U")
    s = Replace(s, ".", "")
    s = Replace(s, ",", "")
    s = Replace(s, "/", "")
    Do While InStr(s, "  ") > 0
        s = Replace(s, "  ", " ")
    Loop
    V08_NormHeader = s
End Function


Private Function V08_LastUsedRow(ByVal ws As Worksheet) As Long
    Dim c As Range

    Set c = ws.Cells.Find(What:="*", _
                          After:=ws.Range("A1"), _
                           LookIn:=xlFormulas, _
                          LookAt:=xlPart, _
                          SearchOrder:=xlByRows, _
                          SearchDirection:=xlPrevious, _
                           MatchCase:=False)

    If c Is Nothing Then
        V08_LastUsedRow = 0
    Else
        V08_LastUsedRow = c.Row
    End If
End Function


Private Function V08_ParseNumber(ByVal v As Variant) As Double
    Dim s As String
    Dim neg As Boolean

    On Error GoTo ErrHandler

    If IsNumeric(v) Then
        V08_ParseNumber = CDbl(v)
        Exit Function
    End If

    s = Trim$(CStr(v))
    If s = "" Then Exit Function

    s = Replace(s, Chr(160), "")
    s = Replace(s, "$", "")
    s = Replace(s, ",", "")
    s = Replace(s, " ", "")

    If Left$(s, 1) = "(" And Right$(s, 1) = ")" Then
        neg = True
        s = Replace(s, "(", "")
        s = Replace(s, ")", "")
    End If

    If IsNumeric(s) Then
        V08_ParseNumber = CDbl(s)
        If neg Then V08_ParseNumber = -V08_ParseNumber
    End If
    Exit Function

ErrHandler:
    V08_ParseNumber = 0
End Function


Private Function V08_GetCellText(ByVal ws As Worksheet, ByVal rowNum As Long, ByVal colNum As Long) As String
    If ws Is Nothing Then Exit Function
    If colNum <= 0 Then Exit Function
    V08_GetCellText = Trim$(CStr(ws.Cells(rowNum, colNum).Value))
End Function


Private Function V08_RowHasData(ByVal ws As Worksheet, ByVal rowNum As Long, ByVal cols As Variant) As Boolean
    Dim i As Long, c As Long

    For i = LBound(cols) To UBound(cols)
        c = CLng(cols(i))
        If c > 0 Then
            If Trim$(CStr(ws.Cells(rowNum, c).Value)) <> "" Then
                V08_RowHasData = True
                 Exit Function
            End If
        End If
    Next i
End Function


Private Function V08_AppendUniqueToken(ByVal existingList As String, ByVal token As String) As String
    Dim arr As Variant
    Dim i As Long
    Dim cleanToken As String

    cleanToken = Trim$(token)
    If cleanToken = "" Then
        V08_AppendUniqueToken = existingList
        Exit Function
     End If

    If Trim$(existingList) = "" Then
        V08_AppendUniqueToken = cleanToken
        Exit Function
    End If

    arr = Split(existingList, " | ")
    For i = LBound(arr) To UBound(arr)
        If StrComp(Trim$(CStr(arr(i))), cleanToken, vbTextCompare) = 0 Then
            V08_AppendUniqueToken = existingList
            Exit Function
        End If
    Next i

    V08_AppendUniqueToken = existingList & " | " & cleanToken
End Function


Private Function V08_NormKey(ByVal v As Variant) As String
    Dim s As String

    s = Trim$(CStr(v))
    If s = "" Then Exit Function

    s = UCase$(s)
    s = Replace(s, Chr(160), "")
    s = Replace(s, " ", "")
    s = Replace(s, "-", "")
    s = Replace(s, "_", "")
    s = Replace(s, ".", "")

    V08_NormKey = s
End Function

Private Function V08_SheetHasText(ByVal ws As Worksheet, ByVal pattern As String) As Boolean
    Dim f As Range
    Set f = ws.Cells.Find(What:=pattern, LookIn:=xlValues, LookAt:=xlPart, MatchCase:=False)
    V08_SheetHasText = Not (f Is Nothing)
End Function


Private Sub V08_DeleteSheetIfExists(ByVal sheetName As String)

    On Error GoTo 0

        Application.DisplayAlerts = True
    End If
End Sub
